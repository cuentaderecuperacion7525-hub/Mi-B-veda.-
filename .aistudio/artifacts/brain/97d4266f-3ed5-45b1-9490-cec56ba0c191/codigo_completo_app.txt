rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper to check user authentication
    function isAuthenticated() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    function isValidUser(data) {
      return data.keys().hasAll(['id', 'firstName', 'email', 'emailVerified', 'createdAt'])
        && data.keys().hasOnly(['id', 'firstName', 'lastName', 'birthDate', 'age', 'email', 'emailVerified', 'photoURL', 'authProvider', 'createdAt', 'updatedAt'])
        && data.id is string
        && data.id.size() <= 128
        && data.firstName is string
        && data.firstName.size() >= 1
        && data.firstName.size() <= 100
        && data.email is string
        && data.email.size() <= 150
        && data.emailVerified is bool
        && data.createdAt is string
        && (!('lastName' in data) || (data.lastName is string && data.lastName.size() <= 100))
        && (!('birthDate' in data) || (data.birthDate is string && data.birthDate.size() <= 20))
        && (!('age' in data) || (data.age is number && data.age >= 0 && data.age <= 130))
        && (!('photoURL' in data) || (data.photoURL is string && data.photoURL.size() <= 2048))
        && (!('authProvider' in data) || (data.authProvider is string && data.authProvider.size() <= 50))
        && (!('updatedAt' in data) || data.updatedAt is string);
    }

    match /users/{userId} {
      allow read: if isOwner(userId);
      allow create: if isOwner(userId) && isValidUser(request.resource.data) && request.resource.data.id == userId;
      allow update: if isOwner(userId) && isValidUser(request.resource.data) && request.resource.data.id == userId;
      allow delete: if isOwner(userId);
    }
  }
}

/**
 * POST /api/verify-email-pin
 * Verifies that the entered 4-digit PIN matches the stored active PIN.
 */
app.post('/api/verify-email-pin', (req: Request, res: Response) => {
  try {
    const { email, pin } = req.body;
    if (!email || !pin) {
      return res.status(400).json({ success: false, error: 'Correo y PIN requeridos.' });
    }

    const cleanEmail = email.trim().toLowerCase();
    const cleanPin = pin.toString().trim();

    const record = verificationCodes.get(cleanEmail);

    if (!record) {
      return res.status(400).json({
        success: false,
        error: 'No se encontró ningún código pendiente para este correo. Solicita uno nuevo.',
      });
    }

    if (Date.now() > record.expiresAt) {
      verificationCodes.delete(cleanEmail);
      return res.status(400).json({
        success: false,
        error: 'El código ha expirado (límite de 10 minutos). Solicita uno nuevo.',
      });
    }

    if (record.pin !== cleanPin) {
      return res.status(400).json({
        success: false,
        error: 'El código de 6 dígitos no coincide. Verifica e intenta de nuevo.',
      });
    }

    // Success: consume the code so it cannot be re-used
    verificationCodes.delete(cleanEmail);
    return res.json({
      success: true,
      message: 'Código verificado correctamente.',
    });
  } catch (err: any) {
    console.error('Error in /api/verify-email-pin:', err);
    return res.status(500).json({
      success: false,
      error: err?.message || 'Error interno al verificar el código.',
    });
  }
});

// Vite dev server mounting or static production serving
async function startServer() {
  const httpServer = http.createServer(app);

  if (!isProd) {
    const { createServer: createViteServer } = await import('vite');
    const vite = await createViteServer({
      server: {
        middlewareMode: true,
        hmr: {
          server: httpServer,
        },
      },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    app.use(express.static(path.join(__dirname, 'dist')));
    app.get('*', (_req, res) => {
      res.sendFile(path.join(__dirname, 'dist', 'index.html'));
    });
  }

  httpServer.listen(PORT, '0.0.0.0', () => {
    console.log(`Server listening on http://0.0.0.0:${PORT}`);
  });
}

startServer();

import React, { useState, useEffect, useRef, useCallback } from 'react';
import { Shield } from 'lucide-react';
import { App as CapacitorApp } from '@capacitor/app';
import { SetupVaultView } from './components/SetupVaultView';
import { LockVaultView } from './components/LockVaultView';
import { MainVaultView } from './components/MainVaultView';
import { InitialRegisterView } from './components/InitialRegisterView';
import { SettingsModal } from './components/SettingsModal';
import { ToastContainer, ToastData } from './components/Toast';
import { openVaultDatabase, clearVaultDatabase, autoPurgeTrash } from './db/vaultDb';
import { initScreenshotBlocker, setScreenshotBlockToastCallback } from './services/screenshotBlockService';
import { initBackButtonListener } from './services/backButtonService';
import { updateService, UpdateCheckResult } from './services/updateService';
import { flipToLockService } from './services/flipToLockService';
import { themeService } from './services/themeService';
import { userProfileService } from './services/userProfileService';
import { testFirestoreConnection } from './services/firebase';
import { UpdateNotificationModal } from './components/UpdateNotificationModal';
import { CURRENT_APP_VERSION } from './version';

export default function App() {
  const [vaultStatus, setVaultStatus] = useState<'checking' | 'register' | 'setup' | 'locked' | 'unlocked'>('checking');
  const [masterKey, setMasterKey] = useState<CryptoKey | null>(null);
  const [sessionPin, setSessionPin] = useState<string | null>(null);
  const [toasts, setToasts] = useState<ToastData[]>([]);
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);
  const [isPrivacyBlurActive, setIsPrivacyBlurActive] = useState(false);

  // Settings
  const [autoLockMinutes, setAutoLockMinutes] = useState<number>(() => {
    const saved = localStorage.getItem('vault_autolock_mins');
    const val = Number(saved);
    return saved !== null && !isNaN(val) ? val : 5;
  });

  const [lockOnTabSwitch, setLockOnTabSwitch] = useState<boolean>(() => {
    const saved = localStorage.getItem('vault_lock_on_tab_switch');
    return saved !== null ? saved === 'true' : true;
  });

  const vaultStatusRef = useRef(vaultStatus);
  useEffect(() => {
    vaultStatusRef.current = vaultStatus;
  }, [vaultStatus]);

  const lockOnTabSwitchRef = useRef(lockOnTabSwitch);
  useEffect(() => {
    lockOnTabSwitchRef.current = lockOnTabSwitch;
  }, [lockOnTabSwitch]);

  const [vaultRefreshTrigger, setVaultRefreshTrigger] = useState<number>(0);

  // Sistema de Actualizaciones
  const [pendingUpdateInfo, setPendingUpdateInfo] = useState<UpdateCheckResult | null>(null);
  const [showUpdateBanner, setShowUpdateBanner] = useState<boolean>(false);
  const [showUpdateModal, setShowUpdateModal] = useState<boolean>(false);
  const [isViewingUpdates, setIsViewingUpdates] = useState<boolean>(() => {
    return typeof window !== 'undefined' && sessionStorage.getItem('vault_updates_menu_active') === 'true';
  });
  const isViewingUpdatesRef = useRef<boolean>(isViewingUpdates);

  useEffect(() => {
    isViewingUpdatesRef.current = isViewingUpdates;
  }, [isViewingUpdates]);

  useEffect(() => {
    const handleUpdateViewState = (e: any) => {
      const isOpen = !!e.detail?.isOpen;
      setIsViewingUpdates(isOpen);
      if (isOpen) {
        setShowUpdateBanner(false);
      }
    };

    window.addEventListener('vault_updates_view_state', handleUpdateViewState);
    return () => window.removeEventListener('vault_updates_view_state', handleUpdateViewState);
  }, []);

  const inactivityTimerRef = useRef<number | null>(null);

  // Toast Helper
  const showToast = useCallback((message: string, type: 'info' | 'success' | 'alert' | 'error' = 'info') => {
    const id = `${Date.now()}_${Math.random()}`;
    setToasts((prev) => [...prev, { id, message, type }]);

    setTimeout(() => {
      setToasts((prev) => prev.filter((t) => t.id !== id));
    }, 3200);
  }, []);

  // Lock Vault Action
  const lockVault = useCallback(() => {
    // Strictly forbid locking while picking files or running backup operations
    if ((window as any).__isPickingFile || (window as any).__isProcessingBackup) {
      return;
    }
    // CRITICAL: NEVER lock if the vault is not in 'unlocked' state!
    // During registration ('register') or setup ('setup'), the user has not finished onboarding;
    // locking would switch them to LockVaultView and corrupt the onboarding flow.
    if (vaultStatusRef.current !== 'unlocked') {
      return;
    }
    try {
      sessionStorage.removeItem('vault_session_pin');
    } catch {}
    setMasterKey(null);
    setSessionPin(null);
    setVaultStatus('locked');
    setIsSettingsOpen(false);
  }, []);

  // Memoized unlock handler with extended post-unlock immunity grace period (20 seconds)
  const handleUnlock = useCallback((key: CryptoKey, pin?: string) => {
    const now = Date.now();
    (window as any).__lastUnlockTime = now;
    (window as any).__lastBiometricUnlockTime = now;
    setMasterKey(key);
    if (pin) {
      setSessionPin(pin);
      try {
        sessionStorage.setItem('vault_session_pin', pin);
      } catch {}
    }
    setVaultStatus('unlocked');
  }, []);

  // Android Back navigation for settings
  const isSettingsOpenRef = useRef(isSettingsOpen);
  isSettingsOpenRef.current = isSettingsOpen;
  const wasSettingsOpenRef = useRef(false);

  useEffect(() => {
    if (isSettingsOpen && !wasSettingsOpenRef.current) {
      window.history.pushState({ appOverlay: 'settings' }, '');
      wasSettingsOpenRef.current = true;
    } else if (!isSettingsOpen && wasSettingsOpenRef.current) {
      wasSettingsOpenRef.current = false;
      if (window.history.state?.appOverlay === 'settings') {
        window.history.back();
      }
    }
  }, [isSettingsOpen]);

  useEffect(() => {
    const handlePopState = () => {
      if (isSettingsOpenRef.current) {
        wasSettingsOpenRef.current = false;
        setIsSettingsOpen(false);
      }
    };
    window.addEventListener('popstate', handlePopState);
    return () => window.removeEventListener('popstate', handlePopState);
  }, []);

  // Check initial vault existence on mount
  useEffect(() => {
    const init = async () => {
      try {
        await openVaultDatabase();
        const isInit = localStorage.getItem('vault_initialized');
        const saltHex = localStorage.getItem('vault_salt');
        const testIvHex = localStorage.getItem('vault_test_iv');
        const testCipherHex = localStorage.getItem('vault_test_cipher');
        const isRegistered = userProfileService.isUserRegistered();

        if (isInit === 'true' && saltHex && testIvHex && testCipherHex) {
          // Ya existen datos y la bóveda está inicializada: salta directo a la pantalla de Autenticación
          setVaultStatus('locked');
        } else if (!isRegistered) {
          // Primera vez que se abre la app (no existen datos guardados): muestra Registro Inicial / Bienvenido
          setVaultStatus('register');
        } else {
          // Usuario registrado pero pendiente de configurar PIN o clave
          setVaultStatus('setup');
        }
        // Auto-limpieza en segundo plano de la papelera (elementos mayores a 30 días)
        autoPurgeTrash(30).catch(() => {});
        testFirestoreConnection().catch(() => {});
      } catch (err) {
        console.error('Initialization error:', err);
        setVaultStatus('setup');
      }
    };

    init();
    initScreenshotBlocker();
    setScreenshotBlockToastCallback(showToast);
    initBackButtonListener();
  }, [showToast]);

  // Bloqueo al Voltear el Teléfono Boca Abajo (Flip to Lock / Face-Down)
  useEffect(() => {
    if (vaultStatus !== 'unlocked') return;

    const unregister = flipToLockService.register(() => {
      showToast('Bóveda bloqueada al voltear el teléfono', 'alert');
      lockVault();
    });

    return () => {
      unregister();
    };
  }, [vaultStatus, lockVault, showToast]);

  // Root back button press notice
  useEffect(() => {
    const handleRootBack = () => {
      showToast('Presiona o desliza atrás de nuevo para salir', 'info');
    };
    const handleRootExit = () => {
      showToast('Saliendo de la bóveda...', 'info');
    };
    window.addEventListener('vault_root_back_press', handleRootBack);
    window.addEventListener('vault_root_exit_allowed', handleRootExit);
    return () => {
      window.removeEventListener('vault_root_back_press', handleRootBack);
      window.removeEventListener('vault_root_exit_allowed', handleRootExit);
    };
  }, [showToast]);

  // Detección automática de actualizaciones al entrar a la aplicación
  useEffect(() => {
    // Si la app acaba de actualizarse, mostrar saludo de confirmación
    if (sessionStorage.getItem('vault_just_updated') === 'true') {
      sessionStorage.removeItem('vault_just_updated');
      setTimeout(() => {
        showToast(`¡Bóveda actualizada con éxito a la versión v${CURRENT_APP_VERSION}!`, 'success');
      }, 700);
    }

    if (!updateService.isAutoCheckEnabled()) return;

    // Comprobar actualizaciones tras 1.5s para no competir con el arranque inicial
    const timer = setTimeout(async () => {
      try {
        const res = await updateService.checkForUpdates();
        if (res.hasUpdate) {
          setPendingUpdateInfo(res);
          if (!updateService.isUpdateDismissed(res.latestVersion)) {
            setShowUpdateBanner(true);
          }
        }
      } catch (e) {
        console.warn('Error al verificar actualizaciones al entrar:', e);
      }
    }, 1500);

    return () => clearTimeout(timer);
  }, [showToast]);

  // Suscribirse a eventos de actualización
  useEffect(() => {
    return updateService.subscribe((res) => {
      if (res.hasUpdate) {
        setPendingUpdateInfo(res);
        if (
          !updateService.isUpdateDismissed(res.latestVersion) &&
          !isViewingUpdatesRef.current &&
          sessionStorage.getItem('vault_updates_menu_active') !== 'true'
        ) {
          setShowUpdateBanner(true);
        }
      }
    });
  }, []);

  // App privacy blur shield and immediate locking when minimizing or switching apps (ONLY when unlocked)
  useEffect(() => {
    let appStateHandle: any = null;

    const isUserPickingFiles = () => {
      if ((window as any).__isPickingFile) return true;
      const startTime = (window as any).__filePickingStartTime || 0;
      if (startTime > 0 && Date.now() - startTime < 180000) return true; // 3 min grace window while picking files
      return false;
    };

    const handleMinimizeOrBackground = () => {
      if (isUserPickingFiles() || (window as any).__isProcessingBackup) {
        return;
      }
      // CRITICAL: NEVER lock if in 'register' or 'setup' mode!
      // The user is legitimately minimizing to retrieve their verification code from Gmail.
      if (vaultStatusRef.current !== 'unlocked') {
        return;
      }
      if (lockOnTabSwitchRef.current) {
        lockVault();
      }
    };

    if (CapacitorApp && typeof CapacitorApp.addListener === 'function') {
      try {
        CapacitorApp.addListener('appStateChange', (state) => {
          if (!isUserPickingFiles()) {
            setIsPrivacyBlurActive(!state.isActive && vaultStatusRef.current === 'unlocked');
          }
          if (!state.isActive) {
            handleMinimizeOrBackground();
          } else {
            setIsPrivacyBlurActive(false);
            window.dispatchEvent(new CustomEvent('vault_app_resumed'));
          }
        }).then((h) => {
          appStateHandle = h;
        }).catch(() => {});
      } catch {}
    }

    const handleDocVisibility = () => {
      const isHidden = document.visibilityState === 'hidden';
      if (!isUserPickingFiles()) {
        setIsPrivacyBlurActive(isHidden && vaultStatusRef.current === 'unlocked');
      }
      if (isHidden) {
        handleMinimizeOrBackground();
      } else {
        setIsPrivacyBlurActive(false);
        window.dispatchEvent(new CustomEvent('vault_app_resumed'));
      }
    };

    const handleWindowBlur = () => {
      if (!isUserPickingFiles() && vaultStatusRef.current === 'unlocked') {
        setIsPrivacyBlurActive(true);
      }
      handleMinimizeOrBackground();
    };

    const handleWindowFocus = () => {
      setIsPrivacyBlurActive(false);
      window.dispatchEvent(new CustomEvent('vault_app_resumed'));
    };

    const handlePageHide = () => {
      handleMinimizeOrBackground();
    };

    document.addEventListener('visibilitychange', handleDocVisibility);
    window.addEventListener('blur', handleWindowBlur);
    window.addEventListener('focus', handleWindowFocus);
    window.addEventListener('pagehide', handlePageHide);

    return () => {
      if (appStateHandle && typeof appStateHandle.remove === 'function') {
        appStateHandle.remove();
      }
      document.removeEventListener('visibilitychange', handleDocVisibility);
      window.removeEventListener('blur', handleWindowBlur);
      window.removeEventListener('focus', handleWindowFocus);
      window.removeEventListener('pagehide', handlePageHide);
    };
  }, [lockVault]);

  // Inactivity auto-lock
  useEffect(() => {
    const mins = Number(autoLockMinutes);
    if (vaultStatus !== 'unlocked' || isNaN(mins) || mins <= 0) {
      if (inactivityTimerRef.current) {
        window.clearTimeout(inactivityTimerRef.current);
        inactivityTimerRef.current = null;
      }
      return;
    }

    const resetTimer = () => {
      if (inactivityTimerRef.current) {
        window.clearTimeout(inactivityTimerRef.current);
      }
      if ((window as any).__isProcessingBackup || (window as any).__isPickingFile) {
        // Pause auto-lock completely during backup generation/restoration or file picking
        return;
      }
      inactivityTimerRef.current = window.setTimeout(() => {
        lockVault();
      }, mins * 60 * 1000);
    };

    resetTimer();

    const activityEvents = ['mousedown', 'mousemove', 'keydown', 'scroll', 'touchstart', 'pointerdown', 'click'];
    activityEvents.forEach((evt) => window.addEventListener(evt, resetTimer, { passive: true }));

    return () => {
      if (inactivityTimerRef.current) {
        window.clearTimeout(inactivityTimerRef.current);
        inactivityTimerRef.current = null;
      }
      activityEvents.forEach((evt) => window.removeEventListener(evt, resetTimer));

        (window as any).__filePickingStartTime = Date.now();

    const handleFileChangeOrCancel = (e: Event) => {
      const target = e.target as HTMLElement;
      if (target && target.tagName === 'INPUT' && (target as HTMLInputElement).type === 'file') {
        setTimeout(() => {
          (window as any).__isPickingFile = false;
        }, 4000);
      }
    };

    window.addEventListener('click', handleFilePickerClick, true);
    window.addEventListener('change', handleFileChangeOrCancel, true);
    window.addEventListener('cancel', handleFileChangeOrCancel, true);

    return () => {
      window.removeEventListener('click', handleFilePickerClick, true);
      window.removeEventListener('change', handleFileChangeOrCancel, true);
      window.removeEventListener('cancel', handleFileChangeOrCancel, true);
    };
  }, []);


  // Panic key: Escape locks immediately
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape' && vaultStatus === 'unlocked' && !isSettingsOpen) {
        lockVault();
      }
    };

  }, [vaultStatus, isSettingsOpen, lockVault]);

      localStorage.removeItem('vault_initialized');
      localStorage.removeItem('vault_pin_length');

      setMasterKey(null);

      }

    } catch (err) {
      console.error('Error resetting vault:', err);
      showToast('Error al restablecer la bóveda', 'error');
    }
  };

  // Preference change listeners
  const handleAutoLockChange = (mins: number) => {
    setAutoLockMinutes(mins);
    localStorage.setItem('vault_autolock_mins', mins.toString());
  };

  const handleLockOnTabSwitchChange = (enabled: boolean) => {
    setLockOnTabSwitch(enabled);
    localStorage.setItem('vault_lock_on_tab_switch', enabled ? 'true' : 'false');
  };

      )}

      )}

      />

      {/* Privacy Blur Shield when minimizing app or switching tasks */}
      {isPrivacyBlurActive && (

/**
 * Decodes a Google ID Token JWT payload without external libraries.
 */
export function decodeGoogleJwt(token: string): {
  sub?: string;
  name?: string;
  given_name?: string;
  family_name?: string;
  email?: string;
  picture?: string;
  email_verified?: boolean;
} | null {
  try {
    const base64Url = token.split('.')[1];
    if (!base64Url) return null;
    const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
    const jsonPayload = decodeURIComponent(
      atob(base64)
        .split('')
        .map((c) => '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2))
        .join('')
    );
    return JSON.parse(jsonPayload);
  } catch (err) {
    console.warn('Failed to parse Google JWT:', err);
    return null;
  }
}

/**

/**
 * Opens the official Google Account Chooser & Consent flow directly via accounts.google.com
 * (Google Identity Services OAuth2 Token Client), avoiding the .firebaseapp.com/__/auth/handler
 * "The requested action is invalid" error on Android Chrome / WebAPK.
 */
export async function signInWithGoogle(): Promise<{
  success: boolean;
  profile?: UserProfile;
  error?: string;
}> {
  return new Promise((resolve) => {
    const oauth2 = window.google?.accounts?.oauth2;

    if (!oauth2) {
      resolve({
        success: false,
        error: 'El servicio de cuentas de Google aún está cargando. Intenta de nuevo en un segundo.',
      });
      return;
    }

    try {
      const tokenClient = oauth2.initTokenClient({
        client_id: GOOGLE_OAUTH_CLIENT_ID,
        scope: 'openid email profile',
        prompt: 'select_account consent',
        callback: async (tokenResponse) => {
          if (tokenResponse.error || !tokenResponse.access_token) {
            if (tokenResponse.error === 'access_denied') {
              resolve({
                success: false,
                error: 'Acceso rechazado. Debes permitir el acceso para continuar.',
              });
            } else {
              resolve({
                success: false,
                error: tokenResponse.error_description || 'Inicio de sesión cancelado.',
              });
            }
            return;
          }

          const result = await signInWithGoogleAccessToken(tokenResponse.access_token);
          resolve(result);
        },
        error_callback: (err) => {
          if (err?.type === 'popup_closed') {
            resolve({
              success: false,
              error: 'Ventana cerrada antes de completar la selección de cuenta.',
            });
          } else if (err?.type === 'popup_failed_to_open') {
            resolve({
              success: false,
              error: 'El navegador bloqueó la ventana emergente de Google. Permite las ventanas emergentes.',
            });
          } else {
            resolve({
              success: false,
              error: 'Operación cancelada.',
            });
          }
        },
      });

      tokenClient.requestAccessToken({ prompt: 'select_account consent' });
    } catch (err: any) {
      console.error('Error launching Google OAuth2 TokenClient:', err);
      resolve({
        success: false,
        error: 'No se pudo abrir el selector oficial de cuentas de Google.',
      });
    }
  });
}

/**
 * Generates a stable deterministic authentication token for local fallback.
 */
function getDeterministicVaultKey(email: string): string {
  const clean = email.trim().toLowerCase();
  let hash = 0;
  for (let i = 0; i < clean.length; i++) {
    hash = (hash << 5) - hash + clean.charCodeAt(i);
    hash |= 0;
  }
  return `Boveda_#${Math.abs(hash)}_SecKey!2026`;
}

/**
 * Fallback: Registers the user with Email and sends a verification link via Firebase Auth.
 */
export async function sendFirebaseVerification(
  email: string,
  firstName: string,
  lastName?: string,
  birthDate?: string,
  age?: number
): Promise<FirebaseRegisterResult> {
  const cleanEmail = email.trim().toLowerCase();
  const password = getDeterministicVaultKey(cleanEmail);

  try {
    let userCredential;

    try {
      userCredential = await createUserWithEmailAndPassword(auth, cleanEmail, password);
    } catch (createErr: any) {
      if (createErr.code === 'auth/email-already-in-use') {
        userCredential = await signInWithEmailAndPassword(auth, cleanEmail, password);
      } else {
        throw createErr;
      }
    }

    const user = userCredential.user;
    await user.reload();

    if (user.emailVerified) {
      await syncUserProfile(user.uid, {
        firstName,
        lastName,
        birthDate,
        age,
        email: cleanEmail,
        emailVerified: true,
        authProvider: 'password',
      });

      return {
        success: true,
        emailSent: false,
        isAlreadyVerified: true,
        message: 'Tu correo ya está verificado oficialmente.',
      };
    }

    await sendEmailVerification(user, {
      url: window.location.origin,
      handleCodeInApp: false,
    });

    return {
      success: true,
      emailSent: true,
      message: `Correo oficial de verificación enviado a ${cleanEmail}`,
    };
  } catch (err: any) {
    console.error('Firebase Auth Verification error:', err);
    let errorMsg = 'Error al enviar el correo de verificación con Firebase.';
    if (err.code === 'auth/invalid-email') {
      errorMsg = 'El formato del correo electrónico no es válido.';
    } else if (err.code === 'auth/too-many-requests') {
      errorMsg = 'Demasiados intentos. Espera unos momentos antes de solicitar otro correo.';
    } else if (err.code === 'auth/operation-not-allowed') {
      errorMsg = 'El registro manual por correo está deshabilitado. Usa el botón "Continuar con Gmail".';
    } else if (err.code === 'auth/network-request-failed') {
      errorMsg = 'Error de conexión. Verifica tu conexión a internet.';
    } else if (err.message) {
      errorMsg = err.message;
    }

    return {
      success: false,
      error: errorMsg,
    };
  }
}

/**
 * Saves profile data to local storage and Firestore.
 */
async function syncUserProfile(
  uid: string,
  profile: {
    firstName: string;
    lastName?: string;
    birthDate?: string;
    age?: number;
    email: string;
    emailVerified: boolean;
    photoURL?: string;
    authProvider?: string;
  }
) {
  userProfileService.saveUserProfile({
    firstName: profile.firstName,
    lastName: profile.lastName,
    birthDate: profile.birthDate,
    age: profile.age,
    email: profile.email,
    emailVerified: profile.emailVerified,
    photoURL: profile.photoURL,
    authProvider: profile.authProvider,
  });

  try {
    const userRef = doc(db, 'users', uid);
    const now = new Date().toISOString();
    const docData: Record<string, any> = {
      id: uid,
      firstName: profile.firstName.trim(),
      email: profile.email.trim().toLowerCase(),
      emailVerified: profile.emailVerified,
      createdAt: now,
      updatedAt: now,
    };

    if (profile.lastName?.trim()) {
      docData.lastName = profile.lastName.trim();
    }
    if (profile.birthDate?.trim()) {
      docData.birthDate = profile.birthDate.trim();
    }
    if (profile.age !== undefined && profile.age >= 0) {
      docData.age = profile.age;
    }
    if (profile.photoURL) {
      docData.photoURL = profile.photoURL;
    }
    if (profile.authProvider) {
      docData.authProvider = profile.authProvider;
    }

    await setDoc(userRef, docData, { merge: true });
  } catch (err) {
    console.warn('Firestore user profile sync warning (offline or permissions):', err);
  }
}

import {
  signInWithGoogle,
  signInWithGoogleCredential,
  GOOGLE_OAUTH_CLIENT_ID,
} from '../services/firebaseAuthService';

interface InitialRegisterViewProps {
  onComplete: () => void;
  showToast: (message: string, type?: 'info' | 'success' | 'alert' | 'error') => void;
}

export const InitialRegisterView: React.FC<InitialRegisterViewProps> = ({
  onComplete,
  showToast,
}) => {
  const [isLoading, setIsLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  // Asegurar que el script oficial de Google Identity Services esté listo en el dispositivo
  useEffect(() => {
    const initGsiIfAvailable = () => {
      if (window.google?.accounts?.id) {
        try {
          window.google.accounts.id.initialize({
            client_id: GOOGLE_OAUTH_CLIENT_ID,
            ux_mode: 'popup',
            auto_select: false,
            cancel_on_tap_outside: true,
            callback: async (response) => {
              if (!response.credential) return;
              setIsLoading(true);
              setErrorMessage(null);
              const res = await signInWithGoogleCredential(response.credential);
              setIsLoading(false);
              if (res.success && res.profile) {
                showToast(`Cuenta verificada: ${res.profile.email}`, 'success');
                onComplete();
              } else if (res.error) {
                setErrorMessage(res.error);
              }
            },
          });
        } catch (err) {
          console.warn('GSI init warning:', err);
        }
      }
    };

    if (window.google?.accounts) {
      initGsiIfAvailable();
    } else {
      const existingScript = document.querySelector(
        'script[src="https://accounts.google.com/gsi/client"]'
      );
      if (!existingScript) {
        const script = document.createElement('script');
        script.src = 'https://accounts.google.com/gsi/client';
        script.async = true;
        script.defer = true;
        script.onload = initGsiIfAvailable;
        document.head.appendChild(script);
      } else {
        existingScript.addEventListener('load', initGsiIfAvailable);
        return () => existingScript.removeEventListener('load', initGsiIfAvailable);
      }
    }
  }, [onComplete, showToast]);

  const waitForGoogleScript = async (timeoutMs = 4000): Promise<boolean> => {
    const start = Date.now();
    while (Date.now() - start < timeoutMs) {
      if (window.google?.accounts?.oauth2) {
        return true;
      }
      await new Promise((r) => setTimeout(r, 150));
    }
    return Boolean(window.google?.accounts?.oauth2);
  };

  const handleContinueWithGmail = async () => {
    if (isLoading) return;
    setIsLoading(true);
    setErrorMessage(null);

    try {
      const ready = await waitForGoogleScript();
      if (!ready) {
        setErrorMessage(
          'No se pudo conectar con Google. Verifica tu conexión a internet e inténtalo de nuevo.'
        );
        setIsLoading(false);
        return;
      }

      const res = await signInWithGoogle();
      if (res.success && res.profile) {
        showToast(`Bienvenido, ${res.profile.firstName}`, 'success');
        onComplete();
        return;
      }

      if (res.error) {
        // No mostrar alerta roja si solo cerró la ventana voluntariamente
        if (
          res.error.includes('cerrada') ||
          res.error.includes('cancelad')
        ) {
          setIsLoading(false);
          return;
        }
        setErrorMessage(res.error);
      }
    } catch {
      setErrorMessage('Error al abrir el selector oficial de cuentas de Google.');
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="min-h-screen w-full bg-[#f9fbfb] flex flex-col justify-between items-center px-6 pt-12 pb-12 select-none font-sans overflow-hidden">
      {/* Ilustración central: Escudo Mejorado Poligonal */}
      <div className="w-full flex-1 flex items-center justify-center">
        <svg
          className="w-full max-w-[290px] sm:max-w-[320px] h-auto"
          viewBox="0 0 360 400"
          fill="none"
          xmlns="http://www.w3.org/2000/svg"
        >
          {/* Halos circulares concéntricos suaves detrás del centro */}
          <circle cx="180" cy="192" r="110" fill="#f1f6f6" />
          <circle cx="180" cy="192" r="78" fill="#e8f1f0" />

          {/* Contorno exterior del escudo */}
          <path
            d="M180 28 L314 82 L314 195 C314 282 256 340 180 374 C104 340 46 282 46 195 L46 82 Z"
            stroke="#9cb0b5"
            strokeWidth="2.2"
            strokeLinecap="round"
            strokeLinejoin="round"
          />

          {/* Red geométrica interna (líneas de conexión) */}
          <g stroke="#9cb0b5" strokeWidth="1.4">
            <line x1="180" y1="28" x2="118" y2="112" />
            <line x1="180" y1="28" x2="242" y2="112" />
            <line x1="46" y1="82" x2="118" y2="112" />
            <line x1="314" y1="82" x2="242" y2="112" />
            <line x1="118" y1="112" x2="242" y2="112" />
            <line x1="118" y1="112" x2="46" y2="195" />
            <line x1="242" y1="112" x2="314" y2="195" />
            <line x1="46" y1="195" x2="138" y2="188" />
            <line x1="314" y1="195" x2="222" y2="188" />
            <line x1="118" y1="112" x2="152" y2="150" />
            <line x1="242" y1="112" x2="208" y2="150" />
            <line x1="152" y1="230" x2="118" y2="280" />
            <line x1="208" y1="230" x2="242" y2="280" />
            <line x1="46" y1="195" x2="118" y2="280" />
            <line x1="314" y1="195" x2="242" y2="280" />
            <line x1="118" y1="280" x2="242" y2="280" />
            <line x1="118" y1="280" x2="180" y2="374" />
            <line x1="242" y1="280" x2="180" y2="374" />
          </g>

          {/* Nodos circulares en los vértices e intersecciones */}
          <g fill="#2d8a7e">
            <circle cx="180" cy="28" r="4.5" />
            <circle cx="46" cy="82" r="3.2" fill="#7fa3a0" />
            <circle cx="314" cy="82" r="3.2" fill="#7fa3a0" />
            <circle cx="118" cy="112" r="4.5" />
            <circle cx="242" cy="112" r="4.5" />
            <circle cx="46" cy="195" r="4.5" />
            <circle cx="314" cy="195" r="4.5" />
            <circle cx="118" cy="280" r="4.5" />
            <circle cx="242" cy="280" r="4.5" />
            <circle cx="180" cy="374" r="4.5" />
          </g>

          {/* Contenedor central blanco redondeado (Squircle) */}
          <rect
            x="136"
            y="146"
            width="88"
            height="88"
            rx="24"
            fill="#ffffff"
            stroke="#227c70"
            strokeWidth="2.2"
          />

          {/* Ícono de candado central en verde petróleo oscuro */}
          <g stroke="#134e48" strokeWidth="3.2" strokeLinecap="round" strokeLinejoin="round">
            <rect x="163" y="187" width="34" height="24" rx="5" fill="none" />
            <path d="M170 187 V179 C170 173.5 174.5 169 180 169 C185.5 169 190 173.5 190 179 V187" />
            <circle cx="180" cy="199" r="2.5" fill="#134e48" />
          </g>
        </svg>
      </div>

      {/* Sección inferior: Textos y Botón Estilizado */}
      <div className="w-full max-w-[340px] flex flex-col items-center">
        <h1 className="text-[28px] sm:text-[30px] font-extrabold text-[#111c24] tracking-tight text-center leading-tight">
          Crea tu bóveda
        </h1>

        <p className="text-[#5a6b75] text-[15px] sm:text-[16px] text-center mt-2.5 leading-relaxed max-w-[260px]">
          Guarda tus fotos, videos y apps en un lugar seguro.
        </p>

        {/* Botón Estético y Proporcionado: Continuar con Gmail */}
        <button
          type="button"
          onClick={handleContinueWithGmail}
          disabled={isLoading}
          className="mt-7 w-full max-w-[270px] h-[50px] px-5 rounded-full bg-gradient-to-r from-[#1b4d48] to-[#235e58] hover:from-[#16403c] hover:to-[#1d504b] active:scale-[0.98] disabled:opacity-70 text-white font-semibold text-[15px] tracking-wide shadow-[0_8px_20px_-4px_rgba(31,81,76,0.38)] border border-white/10 transition-all duration-150 cursor-pointer flex items-center justify-center gap-3"
        >
          {isLoading ? (
            <>
              <Loader2 className="w-5 h-5 animate-spin text-white" />
              <span className="whitespace-nowrap">Conectando con Google...</span>

            <>

import React, { useState } from 'react';
import {
  ShieldCheck,
  KeyRound,
  AlertCircle,
  Sparkles,
  Copy,
  Check,
  Download,
  CheckCircle2,
  Lock,
  Loader2,
  ShieldAlert,
} from 'lucide-react';
import {
  deriveKeyFromPin,
  generateSalt,
  createVerificationPayload,
  bufToHex,
} from '../crypto/vaultCrypto';
import {
  generateRecoveryKey,
  backupMasterKeyWithRecoveryKey,
  downloadRecoveryKeyFile,
  formatRecoveryKeyDisplay,
} from '../services/recoveryKeyService';
import { PinKeypad } from './PinKeypad';

interface SetupVaultViewProps {
  onSetupComplete: (masterKey: CryptoKey, pin?: string) => void;
  showToast: (msg: string, type?: 'info' | 'success' | 'alert' | 'error') => void;
}

export const SetupVaultView: React.FC<SetupVaultViewProps> = ({
  onSetupComplete,
  showToast,
}) => {
  const [step, setStep] = useState<'create' | 'confirm' | 'recovery_key'>('create');
  const [pin, setPin] = useState('');
  const [confirmPin, setConfirmPin] = useState('');
  const [isProcessing, setIsProcessing] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  // Recovery Key states
  const [recoveryKey, setRecoveryKey] = useState('');
  const [copiedKey, setCopiedKey] = useState(false);
  const [downloadedKey, setDownloadedKey] = useState(false);
  const [savedConfirmed, setSavedConfirmed] = useState(false);

  const handleBackToCreate = () => {
    setStep('create');
    setConfirmPin('');
    setErrorMessage(null);
  };

  // Called when PIN is confirmed
  const handlePinConfirmed = (confirmedValue: string) => {
    if (confirmedValue !== pin) {
      setErrorMessage('Los PINs no coinciden. Inténtalo de nuevo.');
      setConfirmPin('');
      return;
    }
    setErrorMessage(null);
    // Generate the master recovery key and display it
    const newKey = generateRecoveryKey();
    setRecoveryKey(newKey);
    setStep('recovery_key');
  };

  // Copies recovery key to clipboard
  const handleCopyKey = () => {
    try {
      navigator.clipboard.writeText(recoveryKey);
      setCopiedKey(true);
      setSavedConfirmed(true);
      showToast('¡Clave de recuperación copiada al portapapeles!', 'success');
      setTimeout(() => setCopiedKey(false), 3000);
    } catch {
      showToast('Selecciona y copia el texto manualmente', 'info');
    }
  };

  // Downloads text backup file
  const handleDownloadKey = () => {
    try {
      downloadRecoveryKeyFile(recoveryKey);
      setDownloadedKey(true);
      setSavedConfirmed(true);
      showToast('Archivo de clave de recuperación descargado', 'success');
    } catch (e) {
      console.error(e);
      showToast('Error al descargar archivo de respaldo', 'error');
    }
  };

  // Finalizes vault cryptographic storage and unlocks
  const finalizeSetup = async () => {
    try {
      setIsProcessing(true);
      setErrorMessage(null);

      // 1. Generate cryptographic salt and derive AES-GCM 256 master key
      const salt = generateSalt();
      const saltHex = bufToHex(salt);
      const masterKey = await deriveKeyFromPin(pin, salt);

      // 2. Create test verification token
      const verification = await createVerificationPayload(masterKey);

      // 3. Store vault state in localStorage
      localStorage.setItem('vault_salt', saltHex);
      localStorage.setItem('vault_test_iv', verification.ivHex);
      localStorage.setItem('vault_test_cipher', verification.cipherHex);
      localStorage.setItem('vault_initialized', 'true');
      localStorage.setItem('vault_pin_length', pin.length.toString());

      try {
        sessionStorage.setItem('vault_session_pin', pin);
      } catch {}

      // 4. Wrap and backup master key with the recovery key
      if (recoveryKey) {
        await backupMasterKeyWithRecoveryKey(masterKey, recoveryKey);
      }

      showToast('¡Bóveda configurada y clave de respaldo activada!', 'success');
      onSetupComplete(masterKey, pin);
    } catch (err) {
      console.error('Error creating vault:', err);
      setErrorMessage('Error al inicializar la bóveda criptográfica');
      setIsProcessing(false);
    }
  };

  return (
    <div className="min-h-screen bg-[#090d16] flex flex-col justify-center items-center p-4">
      {/* Background ambient glow */}
      <div className="fixed inset-0 pointer-events-none flex justify-center items-center overflow-hidden">
        <div className="w-[500px] h-[500px] bg-indigo-600/10 rounded-full blur-[120px]" />
      </div>

      <div className="relative w-full max-w-md bg-[#131c2e]/90 border border-slate-800/90 rounded-3xl p-6 sm:p-8 shadow-2xl backdrop-blur-xl text-center">
        {/* Header Icon */}
        <div className="w-16 h-16 bg-indigo-500/10 border border-indigo-500/30 rounded-2xl flex items-center justify-center mx-auto mb-5 text-indigo-400 shadow-inner shadow-indigo-500/20">
          {step === 'create' ? (
            <KeyRound className="w-8 h-8" />
          ) : step === 'confirm' ? (
            <ShieldCheck className="w-8 h-8 text-emerald-400" />
          ) : (
            <Lock className="w-8 h-8 text-[#00e5ff]" />
          )}
        </div>

        {/* Titles and Descriptions according to step */}
        {step === 'create' && (
          <>
            <h1 className="text-2xl font-bold tracking-tight text-white mb-2">
              Configura tu Bóveda
            </h1>
            <p className="text-sm text-slate-400 mb-6 leading-relaxed">
              Crea un PIN numérico de 4 dígitos. Tus fotos, videos y documentos se cifrarán localmente mediante AES-GCM de 256 bits.
            </p>
          </>
        )}

        {step === 'confirm' && (
          <>
            <h1 className="text-2xl font-bold tracking-tight text-white mb-2">
              Confirma tu PIN
            </h1>
            <p className="text-sm text-slate-400 mb-6 leading-relaxed">
              Ingresa de nuevo el mismo PIN maestro para confirmar la clave de cifrado.
            </p>
          </>
        )}

        {step === 'recovery_key' && (
          <>
            <div className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full bg-[#00e5ff]/10 border border-[#00e5ff]/30 text-[#00e5ff] text-xs font-bold uppercase tracking-wider mb-2">
              <ShieldCheck className="w-3.5 h-3.5" />
              <span>Respaldo de Emergencia</span>
            </div>
            <h1 className="text-2xl font-bold tracking-tight text-white mb-2">
              Tu Clave de <span className="text-[#00e5ff]">Respaldo Maestra</span>
            </h1>
            <p className="text-xs text-slate-400 mb-5 leading-relaxed">
              Guarda esta clave en un lugar seguro. Si alguna vez olvidas tu PIN, esta clave es la que te permitirá recuperarlo <strong>sin perder ninguna de tus fotos</strong>.
            </p>
          </>
        )}

        {/* Step indicator */}
        <div className="flex items-center justify-center gap-2 mb-6">
          <span
            className={`h-1.5 rounded-full transition-all duration-300 ${
              step === 'create' ? 'w-8 bg-indigo-500' : 'w-2 bg-slate-700'
            }`}
          />
          <span
            className={`h-1.5 rounded-full transition-all duration-300 ${
              step === 'confirm' ? 'w-8 bg-indigo-500' : 'w-2 bg-slate-700'
            }`}
          />
          <span
            className={`h-1.5 rounded-full transition-all duration-300 ${
              step === 'recovery_key' ? 'w-8 bg-[#00e5ff]' : 'w-2 bg-slate-700'
            }`}
          />
        </div>

        {/* STEP 1: CREATE PIN */}
        {step === 'create' && (
          <div>
            <PinKeypad
              pin={pin}
              onPinChange={(val) => {
                setPin(val);
                setErrorMessage(null);
                if (val.length === 4) {
                  setTimeout(() => {
                    setStep('confirm');
                  }, 120);
                }
              }}
              maxDigits={4}
              disabled={isProcessing}
            />

            {errorMessage && (
              <div className="mt-4 flex items-center justify-center gap-1.5 text-rose-400 text-xs bg-rose-500/10 py-2 px-3 rounded-xl border border-rose-500/20">
                <AlertCircle className="w-4 h-4 shrink-0" />
                <span>{errorMessage}</span>
              </div>
            )}
          </div>
        )}

        {/* STEP 2: CONFIRM PIN */}
        {step === 'confirm' && (
          <div>
            <PinKeypad
              pin={confirmPin}
              onPinChange={(val) => {
                setConfirmPin(val);
                setErrorMessage(null);
                if (val.length === 4) {
                  handlePinConfirmed(val);
                }
              }}
              maxDigits={4}
              disabled={isProcessing}
            />

            {errorMessage && (
              <div className="mt-4 flex items-center justify-center gap-1.5 text-rose-400 text-xs bg-rose-500/10 py-2 px-3 rounded-xl border border-rose-500/20">
                <AlertCircle className="w-4 h-4 shrink-0" />
                <span>{errorMessage}</span>
              </div>
            )}

            <div className="mt-5 flex justify-center">
              <button
                type="button"
                onClick={handleBackToCreate}
                disabled={isProcessing}
                className="py-2.5 px-6 bg-slate-800/80 hover:bg-slate-700 text-slate-300 text-xs font-medium rounded-xl border border-slate-700/80 transition-colors cursor-pointer"
              >
                Volver a cambiar PIN
              </button>
            </div>
          </div>
        )}

        {/* STEP 3: RECOVERY KEY DISPLAY */}
        {step === 'recovery_key' && (
          <div className="space-y-4 text-left">
            {/* Clave Card */}
            <div className="bg-[#0b101d] border-2 border-[#00e5ff]/40 rounded-2xl p-4 shadow-xl shadow-cyan-500/10 text-center">
              <p className="text-[10px] uppercase font-bold tracking-widest text-[#00e5ff] mb-1.5">
                Clave de Seguridad Personal (16 dígitos)
              </p>
              <div className="py-2 px-3 bg-black/60 rounded-xl border border-slate-800">
                <span className="text-xl sm:text-2xl font-black font-mono tracking-widest text-white selection:bg-[#00e5ff] selection:text-black">
                  {formatRecoveryKeyDisplay(recoveryKey)}
                </span>
              </div>

              {/* Action Buttons: Copiar y Descargar */}
              <div className="grid grid-cols-2 gap-2 mt-3">
                <button
                  type="button"
                  onClick={handleCopyKey}
                  className="flex items-center justify-center gap-1.5 py-2.5 px-3 bg-[#00e5ff]/20 hover:bg-[#00e5ff]/30 text-[#00e5ff] text-xs font-semibold rounded-xl border border-[#00e5ff]/40 transition-colors cursor-pointer active:scale-95"
                >
                  {copiedKey ? (
                    <>
                      <Check className="w-4 h-4 text-emerald-400" />
                      <span>¡Copiada!</span>
                    </>
                  ) : (
                    <>
                      <Copy className="w-4 h-4" />
                      <span>Copiar Clave</span>
                    </>
                  )}
                </button>

                <button
                  type="button"
                  onClick={handleDownloadKey}
                  className="flex items-center justify-center gap-1.5 py-2.5 px-3 bg-slate-800 hover:bg-slate-700 text-slate-200 text-xs font-semibold rounded-xl border border-slate-700 transition-colors cursor-pointer active:scale-95"
                >
                  {downloadedKey ? (
                    <>
                      <CheckCircle2 className="w-4 h-4 text-emerald-400" />
                      <span>Descargado</span>
                    </>
                  ) : (
                    <>
                      <Download className="w-4 h-4" />
                      <span>Descargar .txt</span>
                    </>
                  )}
                </button>
              </div>
            </div>

            {/* Checkbox confirmation */}
            <label className="flex items-start gap-2.5 p-3 rounded-xl bg-slate-900/60 border border-slate-800/80 cursor-pointer select-none">
              <input
                type="checkbox"
                checked={savedConfirmed}
                onChange={(e) => setSavedConfirmed(e.target.checked)}
                className="mt-0.5 w-4 h-4 rounded text-[#00e5ff] border-slate-700 focus:ring-0 focus:ring-offset-0 bg-slate-800 accent-[#00e5ff] cursor-pointer"
              />
              <span className="text-xs text-slate-300 leading-snug">
                He copiado, anotado o guardado esta clave en un lugar seguro. Entiendo que es necesaria si olvido mi PIN.
              </span>
            </label>

            {errorMessage && (
              <div className="flex items-center gap-2 text-rose-400 text-xs bg-rose-500/10 p-3 rounded-xl border border-rose-500/20">
                <AlertCircle className="w-4 h-4 shrink-0" />
                <span>{errorMessage}</span>
              </div>
            )}

            {/* Main Action Button */}
            <button
              type="button"
              onClick={finalizeSetup}
              disabled={isProcessing || (!savedConfirmed && !copiedKey && !downloadedKey)}
              className="w-full py-3.5 text-white font-bold rounded-xl text-sm transition-all flex items-center justify-center gap-2 shadow-lg cursor-pointer active:scale-95 disabled:opacity-40 disabled:cursor-not-allowed"
              style={{
                background: 'linear-gradient(135deg, var(--theme-primary), #2563eb)',
                boxShadow: '0 4px 14px -2px var(--theme-glow)',
              }}
            >
