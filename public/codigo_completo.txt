package com.calculadora.boveda.privada;

import android.Manifest;
import android.content.pm.PackageManager;
import android.os.Build;
import android.os.Bundle;
import android.view.WindowManager;
import android.webkit.PermissionRequest;
import androidx.core.app.ActivityCompat;
import androidx.core.content.ContextCompat;
import com.getcapacitor.BridgeActivity;
import com.getcapacitor.BridgeWebChromeClient;
import java.util.ArrayList;
import java.util.List;

public class MainActivity extends BridgeActivity {
    private static final int PERMISSION_REQ_CODE = 1001;

    @Override
    public void onCreate(Bundle savedInstanceState) {
        registerPlugin(VaultBiometricsPlugin.class);
        registerPlugin(VaultSecurityPlugin.class);
        registerPlugin(VaultMediaPlugin.class);
        super.onCreate(savedInstanceState);

        // Bloqueo total a nivel de sistema operativo para capturas y grabaciones de pantalla
        getWindow().setFlags(
            WindowManager.LayoutParams.FLAG_SECURE,
            WindowManager.LayoutParams.FLAG_SECURE
        );

        // Solicitar permisos necesarios desde el inicio (Cámara + Acceso Multimedia para MediaStore)
        checkAndRequestAppPermissions();

        // Concesión automática de permisos de cámara al WebView interno
        if (getBridge() != null && getBridge().getWebView() != null) {
            getBridge().getWebView().setWebChromeClient(new BridgeWebChromeClient(getBridge()) {
                @Override
                public void onPermissionRequest(final PermissionRequest request) {
                    request.grant(request.getResources());
                }
            });
        }
    }

    private void checkAndRequestAppPermissions() {
        List<String> neededPermissions = new ArrayList<>();

        if (ContextCompat.checkSelfPermission(this, Manifest.permission.CAMERA) != PackageManager.PERMISSION_GRANTED) {
            neededPermissions.add(Manifest.permission.CAMERA);
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            // Android 13+ (API 33+)
            if (ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_IMAGES) != PackageManager.PERMISSION_GRANTED) {
                neededPermissions.add(Manifest.permission.READ_MEDIA_IMAGES);
            }
            if (ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_VIDEO) != PackageManager.PERMISSION_GRANTED) {
                neededPermissions.add(Manifest.permission.READ_MEDIA_VIDEO);
            }
            if (ContextCompat.checkSelfPermission(this, Manifest.permission.READ_MEDIA_AUDIO) != PackageManager.PERMISSION_GRANTED) {
                neededPermissions.add(Manifest.permission.READ_MEDIA_AUDIO);
            }
        } else {
            // Android 12 y anteriores
            if (ContextCompat.checkSelfPermission(this, Manifest.permission.READ_EXTERNAL_STORAGE) != PackageManager.PERMISSION_GRANTED) {
                neededPermissions.add(Manifest.permission.READ_EXTERNAL_STORAGE);
            }
            if (Build.VERSION.SDK_INT <= Build.VERSION_CODES.Q) {
                if (ContextCompat.checkSelfPermission(this, Manifest.permission.WRITE_EXTERNAL_STORAGE) != PackageManager.PERMISSION_GRANTED) {
                    neededPermissions.add(Manifest.permission.WRITE_EXTERNAL_STORAGE);
                }
            }
        }

        if (!neededPermissions.isEmpty()) {
            ActivityCompat.requestPermissions(this, neededPermissions.toArray(new String[0]), PERMISSION_REQ_CODE);
        }
    }
}

package com.calculadora.boveda.privada;

import android.app.Activity;
import android.app.PendingIntent;
import android.content.ContentResolver;
import android.content.ContentUris;
import android.content.Intent;
import android.content.IntentSender;
import android.database.Cursor;
import android.net.Uri;
import android.os.Build;
import android.provider.MediaStore;
import androidx.activity.result.ActivityResult;
import androidx.activity.result.IntentSenderRequest;
import com.getcapacitor.JSArray;
import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.ActivityCallback;
import com.getcapacitor.annotation.CapacitorPlugin;
import java.util.ArrayList;
import java.util.List;
import org.json.JSONObject;

@CapacitorPlugin(name = "VaultMedia")
public class VaultMediaPlugin extends Plugin {

    @PluginMethod
    public void openGalleryApp(PluginCall call) {
        try {
            Intent intent = new Intent(Intent.ACTION_VIEW);
            intent.setDataAndType(MediaStore.Images.Media.EXTERNAL_CONTENT_URI, "image/*");
            intent.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            getContext().startActivity(intent);

            JSObject ret = new JSObject();
            ret.put("success", true);
            call.resolve(ret);
        } catch (Exception e) {
            try {
                Intent fallback = Intent.makeMainSelectorActivity(Intent.ACTION_MAIN, Intent.CATEGORY_APP_GALLERY);
                fallback.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
                getContext().startActivity(fallback);
                JSObject ret = new JSObject();
                ret.put("success", true);
                call.resolve(ret);
            } catch (Exception ex) {
                JSObject ret = new JSObject();
                ret.put("success", false);
                ret.put("error", ex.getMessage());
                call.resolve(ret);
            }
        }
    }

    /**
     * Borrado por lotes de archivos multimedia en Android MediaStore.
     * Soporta recibir items con { filename, size, lastModified, mimeType, uri }.
     */
    @PluginMethod
    public void deleteBatchFromGallery(PluginCall call) {
        JSArray itemsArray = call.getArray("items");
        if (itemsArray == null || itemsArray.length() == 0) {
            String filename = call.getString("filename");
            if (filename != null && !filename.isEmpty()) {
                itemsArray = new JSArray();
                JSObject single = new JSObject();
                single.put("filename", filename);
                if (call.getData().has("size")) single.put("size", call.getInt("size"));
                if (call.getData().has("lastModified")) single.put("lastModified", call.getLong("lastModified"));
                if (call.getData().has("uri")) single.put("uri", call.getString("uri"));
                itemsArray.put(single);
            } else {
                call.reject("Se requiere array 'items' o parámetro 'filename'");
                return;
            }
        }

        try {
            ContentResolver resolver = getContext().getContentResolver();
            List<Uri> urisToDelete = new ArrayList<>();

            Uri[] baseUris = new Uri[]{
                MediaStore.Images.Media.EXTERNAL_CONTENT_URI,
                MediaStore.Video.Media.EXTERNAL_CONTENT_URI,
                MediaStore.Audio.Media.EXTERNAL_CONTENT_URI,
                MediaStore.Files.getContentUri("external")
            };

            for (int i = 0; i < itemsArray.length(); i++) {
                JSONObject item = itemsArray.getJSONObject(i);
                String uriStr = item.optString("uri", null);
                String filename = item.optString("filename", item.optString("name", null));
                long size = item.optLong("size", -1);
                long lastModified = item.optLong("lastModified", -1);

                if (uriStr != null && !uriStr.isEmpty() && uriStr.startsWith("content://")) {
                    Uri parsed = Uri.parse(uriStr);
                    if (!urisToDelete.contains(parsed)) {
                        urisToDelete.add(parsed);
                    }
                    continue;
                }

                if (filename == null || filename.isEmpty()) continue;

                Uri foundUri = findMediaStoreUri(resolver, baseUris, filename, size, lastModified);
                if (foundUri != null && !urisToDelete.contains(foundUri)) {
                    urisToDelete.add(foundUri);
                }
            }

            if (urisToDelete.isEmpty()) {
                JSObject ret = new JSObject();
                ret.put("success", false);
                ret.put("matchedCount", 0);
                ret.put("requested", false);
                ret.put("message", "No se encontraron los archivos en MediaStore");
                call.resolve(ret);
                return;
            }

            // En Android 11+ (API 30+) MediaStore.createDeleteRequest pide permiso al usuario con el diálogo nativo del sistema
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                PendingIntent pi = null;
                try {
                    pi = MediaStore.createDeleteRequest(resolver, urisToDelete);
                } catch (Exception exBatch) {
                    List<Uri> imageOnly = new ArrayList<>();
                    for (Uri u : urisToDelete) {
                        if (u.toString().contains("images")) imageOnly.add(u);
                    }
                    if (!imageOnly.isEmpty()) {
                        try {
                            pi = MediaStore.createDeleteRequest(resolver, imageOnly);
                        } catch (Exception ignored) {}
                    }
                }

                if (pi != null) {
                    IntentSender intentSender = pi.getIntentSender();
                    try {
                        getActivity().startIntentSenderForResult(intentSender, 2002, null, 0, 0, 0, null);
                        JSObject ret = new JSObject();
                        ret.put("success", true);
                        ret.put("requested", true);
                        ret.put("urisCount", urisToDelete.size());
                        call.resolve(ret);
                        return;
                    } catch (Exception exSender) {
                        call.reject("No se pudo iniciar el diálogo de eliminación: " + exSender.getMessage());
                        return;
                    }
                } else {
                    JSObject ret = new JSObject();
                    ret.put("success", false);
                    ret.put("requested", false);
                    call.resolve(ret);
                    return;
                }
            } else {
                int deletedCount = 0;
                for (Uri u : urisToDelete) {
                    try {
                        deletedCount += resolver.delete(u, null, null);
                    } catch (Exception ex) {
                        // continuar
                    }
                }
                JSObject ret = new JSObject();
                ret.put("success", deletedCount > 0);
                ret.put("requested", true);
                ret.put("deletedCount", deletedCount);
                call.resolve(ret);
            }
        } catch (Exception e) {
            JSObject ret = new JSObject();
            ret.put("success", false);
            ret.put("requested", false);
            ret.put("error", e.getMessage());
            call.resolve(ret);
        }
    }

    @PluginMethod
    public void deleteFromGallery(PluginCall call) {
        deleteBatchFromGallery(call);
    }

    /**
     * Búsqueda flexible y robusta en MediaStore:
     * 1) Por Nombre y Tamaño exactos
     * 2) Por Nombre exacto (insensible a mayúsculas/minúsculas)
     * 3) Por ID numérico extraído de Photo Picker / DocumentsUI
     * 4) Por prefijo del nombre
     * 5) Por Tamaño exacto (ordenado por más reciente)
     */
    private Uri findMediaStoreUri(ContentResolver resolver, Uri[] baseUris, String filename, long size, long lastModified) {
        for (Uri baseUri : baseUris) {
            String[] projection = new String[]{
                MediaStore.MediaColumns._ID,
                MediaStore.MediaColumns.DISPLAY_NAME,
                MediaStore.MediaColumns.SIZE,
                MediaStore.MediaColumns.DATE_MODIFIED
            };

            // 1. Intento por Nombre y Tamaño exactos
            if (size > 0 && filename != null && !filename.isEmpty()) {
                String selection = MediaStore.MediaColumns.DISPLAY_NAME + " = ? AND " + MediaStore.MediaColumns.SIZE + " = ?";
                String[] selectionArgs = new String[]{filename, String.valueOf(size)};
                try (Cursor cursor = resolver.query(baseUri, projection, selection, selectionArgs, MediaStore.MediaColumns._ID + " DESC")) {
                    if (cursor != null && cursor.moveToFirst()) {
                        long id = cursor.getLong(cursor.getColumnIndexOrThrow(MediaStore.MediaColumns._ID));
                        return ContentUris.withAppendedId(baseUri, id);
                    }
                } catch (Exception ignored) {}
            }

            // 2. Intento por Nombre exacto (sin importar mayúsculas/minúsculas)
            if (filename != null && !filename.isEmpty()) {
                String selectionName = MediaStore.MediaColumns.DISPLAY_NAME + " = ? OR LOWER(" + MediaStore.MediaColumns.DISPLAY_NAME + ") = LOWER(?)";
                String[] selectionArgsName = new String[]{filename, filename};
                try (Cursor cursor = resolver.query(baseUri, projection, selectionName, selectionArgsName, MediaStore.MediaColumns._ID + " DESC")) {
                    if (cursor != null && cursor.moveToFirst()) {
                        long id = cursor.getLong(cursor.getColumnIndexOrThrow(MediaStore.MediaColumns._ID));
                        return ContentUris.withAppendedId(baseUri, id);
                    }
                } catch (Exception ignored) {}
            }

            // 3. Intento por ID numérico si el selector devolvió "image:12345" o similar
            if (filename != null && filename.matches(".*\\d{3,}.*")) {
                try {
                    java.util.regex.Matcher matcher = java.util.regex.Pattern.compile("\\d{3,}").matcher(filename);
                    if (matcher.find()) {
                        long numericId = Long.parseLong(matcher.group());
                        Uri candidate = ContentUris.withAppendedId(baseUri, numericId);
                        try (Cursor cursor = resolver.query(candidate, new String[]{MediaStore.MediaColumns._ID}, null, null, null)) {
                            if (cursor != null && cursor.moveToFirst()) {
                                return candidate;
                            }
                        } catch (Exception ignored) {}
                    }
                } catch (Exception ignored) {}
            }

            // 4. Intento sin extensión si fue cambiado (ej: "IMG_20261009_072505")
            if (filename != null && filename.contains(".")) {
                String baseName = filename.substring(0, filename.lastIndexOf('.'));
                if (baseName.length() >= 4) {
                    String selectionLike = MediaStore.MediaColumns.DISPLAY_NAME + " LIKE ?";
                    String[] selectionArgsLike = new String[]{baseName + "%"};
                    try (Cursor cursor = resolver.query(baseUri, projection, selectionLike, selectionArgsLike, MediaStore.MediaColumns._ID + " DESC")) {
                        if (cursor != null && cursor.moveToFirst()) {
                            long id = cursor.getLong(cursor.getColumnIndexOrThrow(MediaStore.MediaColumns._ID));
                            return ContentUris.withAppendedId(baseUri, id);
                        }
                    } catch (Exception ignored) {}
                }
            }

            // 5. Intento por tamaño exacto (si el tamaño es > 10KB), tomando el archivo más reciente
            if (size > 10240) {
                String selectionSize = MediaStore.MediaColumns.SIZE + " = ?";
                String[] selectionArgsSize = new String[]{String.valueOf(size)};
                try (Cursor cursor = resolver.query(baseUri, projection, selectionSize, selectionArgsSize, MediaStore.MediaColumns._ID + " DESC")) {
                    if (cursor != null && cursor.moveToFirst()) {
                        long id = cursor.getLong(cursor.getColumnIndexOrThrow(MediaStore.MediaColumns._ID));
                        return ContentUris.withAppendedId(baseUri, id);
                    }
                } catch (Exception ignored) {}
            }
        }
        return null;
    }
}

import { Capacitor, registerPlugin } from '@capacitor/core';

export interface NativeMediaItemIdentifier {
  filename: string;
  size?: number;
  lastModified?: number;
  mimeType?: string;
  uri?: string;
}

export interface NativeVaultMediaPlugin {
  /**
   * Elimina un archivo multimedia de la galería pública usando MediaStore en Android
   */
  deleteFromGallery(options: {
    uri?: string;
    filename?: string;
    size?: number;
    lastModified?: number;
  }): Promise<{ success: boolean; message?: string }>;

  /**
   * Elimina por lotes un conjunto de archivos multimedia de la galería pública usando un único
   * diálogo del sistema en Android 11+ (MediaStore.createDeleteRequest).
   */
  deleteBatchFromGallery(options: {
    items: NativeMediaItemIdentifier[];
  }): Promise<{
    success: boolean;
    requested?: boolean;
    confirmedByUser?: boolean;
    matchedCount?: number;
    deletedCount?: number;
    message?: string;
    error?: string;
  }>;

  /**
   * Abre la aplicación nativa de Galería del dispositivo
   */
  openGalleryApp(): Promise<{ success: boolean; error?: string }>;
}

// Registro del plugin nativo de Capacitor (se enlaza en Android con VaultMediaPlugin.java)
export const NativeVaultMedia = registerPlugin<NativeVaultMediaPlugin>('VaultMedia');

/**
 * Retorna true si la app se está ejecutando como un APK nativo en Android o iOS
 */
export const isNativeApp = (): boolean => {
  return Capacitor.isNativePlatform();
};

/**
 * Abre la app nativa de galería de fotos de Android o iOS de manera directa
 */
export const openNativeGallery = async (): Promise<boolean> => {
  if (isNativeApp()) {
    try {
      const res = await NativeVaultMedia.openGalleryApp();
      if (res && res.success) return true;
    } catch (e) {
      console.warn('Could not launch gallery via native plugin:', e);
    }
  }

  try {
    const isAndroid = /Android/i.test(navigator.userAgent);
    if (isAndroid) {
      // Intent standard de Android para ver la Galería multimedia
      window.location.href = 'intent:#Intent;action=android.intent.action.VIEW;type=image/*;end';
    } else {
      window.open('photos-redirect://', '_blank');
    }
  } catch {}
  return true;
};

/**
 * Abre el explorador de archivos / gestor de descargas en Android
 */
export const openNativeFileManager = async (): Promise<boolean> => {
  try {
    const isAndroid = /Android/i.test(navigator.userAgent);
    if (isAndroid) {
      // Abre el gestor de archivos / descargas de Android
      window.location.href = 'intent:#Intent;action=android.intent.action.GET_CONTENT;type=*/*;category=android.intent.category.OPENABLE;end';
    } else {
      window.open('shareddocuments://', '_blank');
    }
  } catch {}
  return true;
};

/**
 * Función unificada para borrar un archivo individual de la galería pública.
 */
export const requestNativeGalleryDeletion = async (
  filename: string,
  uri?: string,
  size?: number,
  lastModified?: number
): Promise<{ handledNatively: boolean; success: boolean }> => {
  if (!isNativeApp()) {
    return { handledNatively: false, success: false };
  }

  try {
    const result = await NativeVaultMedia.deleteFromGallery({ uri, filename, size, lastModified });
    return { handledNatively: true, success: result.success };
  } catch (error) {
    console.warn('Native deletion not handled or rejected by user:', error);
    return { handledNatively: true, success: false };
  }
};

/**
 * Función por lotes para borrar múltiples archivos originales con un solo diálogo nativo.
 */
export const requestNativeBatchGalleryDeletion = async (
  items: NativeMediaItemIdentifier[]
): Promise<{ handledNatively: boolean; success: boolean; requested?: boolean }> => {
  if (!isNativeApp() || !items || items.length === 0) {
    return { handledNatively: false, success: false };
  }

  try {

      success: result.success,
      requested: result.requested,
    };
  } catch (error) {
    console.warn('Native batch deletion error:', error);
    return { handledNatively: true, success: false };

        return { show: true, days, hasNeverBackedUp: false };
      }

      return { show: true, days: 7, hasNeverBackedUp: true };
    }
    return { show: false, days: 0, hasNeverBackedUp: false };
  });

    }
  };

  // Drag and Drop handlers
  const handleDragOver = (e: React.DragEvent) => {
    e.preventDefault();
    setIsDraggingOver(true);
  };

  const handleDragLeave = (e: React.DragEvent) => {
    e.preventDefault();
    setIsDraggingOver(false);
  };

  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault();
    setIsDraggingOver(false);
    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
      const filesArray = Array.from(e.dataTransfer.files);
      processFilesForVault(filesArray);
    }
  };

  // Delete & Favorite
  const handleDeleteItem = async (id: string) => {
    await deleteEncryptedItem(id);
    setItems((prev) => prev.filter((p) => p.id !== id));
    setActiveViewerItem(null);
  };

  const handleToggleFavorite = async (id: string, isFav: boolean) => {
    await updateItemMetadata(id, { isFavorite: isFav });
    setItems((prev) =>
      prev.map((p) =>
        p.id === id ? { ...p, record: { ...p.record, isFavorite: isFav } } : p
      )
    );
  };

  // Batch actions
  const toggleSelectItem = (id: string, e?: React.MouseEvent | React.TouchEvent) => {
    if (e && typeof e.stopPropagation === 'function') {
      e.stopPropagation();
    }
    setSelectedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) {
        next.delete(id);
      } else {
        next.add(id);
      }
      return next;
    });
  };

  const handleSelectAll = () => {
    if (selectedIds.size === filteredItems.length) {
      setSelectedIds(new Set());
    } else {
      setSelectedIds(new Set(filteredItems.map((p) => p.id)));
    }
  };

  const handleBatchDelete = () => {
    if (selectedIds.size === 0) return;
    setShowBatchDeleteModal(true);
  };

  const confirmBatchDelete = async () => {
    if (selectedIds.size === 0) return;
    setIsDeletingBatch(true);
    try {
      const idsToDelete = Array.from(selectedIds);
      await deleteMultipleEncryptedItems(idsToDelete);
      showToast(`${idsToDelete.length} elemento(s) eliminado(s) de la bóveda`, 'success');
      setItems((prev) => prev.filter((p) => !selectedIds.has(p.id)));
      setSelectedIds(new Set());
      setIsSelectMode(false);
      setShowBatchDeleteModal(false);
    } catch (err) {
      console.error('Error deleting items:', err);
      showToast('Error al eliminar elementos', 'error');
    } finally {
      setIsDeletingBatch(false);
    }
  };

  const handleCardPointerDown = (id: string) => {
    isLongPressTriggered.current = false;
    longPressTimeoutRef.current = setTimeout(() => {
      isLongPressTriggered.current = true;
      if (typeof navigator !== 'undefined' && 'vibrate' in navigator) {
        try {
          navigator.vibrate(40);
        } catch {}
      }
      setIsSelectMode(true);
      setSelectedIds((prev) => new Set(prev).add(id));
    }, 400);
  };

  const handleCardPointerUp = () => {
    if (longPressTimeoutRef.current) {
      clearTimeout(longPressTimeoutRef.current);
      longPressTimeoutRef.current = null;
    }
  };

  const handleBatchExport = () => {
    if (selectedIds.size === 0) return;
    const selected = items.filter((p) => selectedIds.has(p.id));

    selected.forEach((p, idx) => {
      setTimeout(() => {
        const link = document.createElement('a');
        link.href = p.blobUrl;
        link.download = p.record.name;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
      }, idx * 250);
    });

    showToast(`Exportando ${selected.length} elemento(s)...`, 'success');
    setIsSelectMode(false);
    setSelectedIds(new Set());
  };

  // Counts by category
  const counts = useMemo(() => {
    let photos = 0;
    let videos = 0;
    let audios = 0;
    let apps = 0;
    let favs = 0;
    let totalBytes = 0;

    items.forEach((item) => {
      totalBytes += item.record.size || 0;
      if (item.record.category === 'photo') photos++;
      else if (item.record.category === 'video') videos++;
      else if (item.record.category === 'audio') audios++;
      else apps++;

      if (item.record.isFavorite) favs++;
    });

    return {
      photos,
      videos,
      audios,
      apps,
      favs,
      totalBytes,
      totalMB: (totalBytes / (1024 * 1024)).toFixed(1),
    };
  }, [items]);

  // Filtered items
  const filteredItems = useMemo(() => {
    return items.filter((item) => {
      if (activeTab === 'favorites') {
        if (!item.record.isFavorite) return false;
      } else if (activeTab !== 'all') {
        if (item.record.category !== activeTab) return false;

        // Subfolder filter
        if (activeSubfolder !== null) {
          const itemAlbum = item.record.album || 'General';
          if (itemAlbum !== activeSubfolder) return false;
        }
      }

      if (searchQuery.trim()) {
        const q = searchQuery.toLowerCase();
        const matchesName = (item.record.name || '').toLowerCase().includes(q);
        const matchesDate = new Date(item.record.timestamp).toLocaleDateString().includes(q);
        const matchesAlbum = (item.record.album || '').toLowerCase().includes(q);
        return matchesName || matchesDate || matchesAlbum;
      }

      return true;
    });
  }, [items, activeTab, activeSubfolder, searchQuery]);

  const formatSize = (bytes: number) => {
    if (!bytes) return '0 KB';
    if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
    return (bytes / (1024 * 1024)).toFixed(1) + ' MB';
  };

  return (
    <div
      onTouchStart={handleTouchStart}
      onTouchEnd={handleTouchEnd}
      onDragOver={handleDragOver}
      onDragLeave={handleDragLeave}
      onDrop={handleDrop}
      className="min-h-screen bg-[#090d16] text-slate-100 flex flex-col relative select-none pb-28 w-full max-w-full"
    >
      {/* Drag & Drop Visual Overlay */}
      {isDraggingOver && (
        <div className="fixed inset-0 z-50 bg-[#0a121d]/90 backdrop-blur-md border-4 border-dashed border-[#00e5ff] flex flex-col items-center justify-center pointer-events-none transition-all">
          <UploadCloud className="w-16 h-16 text-[#00e5ff] animate-bounce mb-3" />
          <h3 className="text-xl font-bold text-white">Suelta fotos, videos o aplicaciones aquí</h3>
          <p className="text-sm text-cyan-200">Se cifrarán con AES-GCM de 256 bits inmediatamente</p>
        </div>
      )}

      {/* Screen 2: Importar Tab */}
      {activeNavTab === 'import' ? (
        <ImportVaultView
          masterKey={masterKey}
          onImportComplete={async (imported) => {
            await loadVaultItems();
            setActiveNavTab('vault');
            if (imported && imported.length > 0) {
              const photoCount = imported.filter((i) => i.category === 'photo').length;
              const videoCount = imported.filter((i) => i.category === 'video').length;
              const appCount = imported.filter((i) => i.category === 'app').length;
              const audioCount = imported.filter((i) => i.category === 'audio').length;

              const media = imported.filter((i) => i.category === 'photo' || i.category === 'video');
              if (media.length > 0) {
                setRecentImportedItems(media.map((m) => ({ name: m.name, size: m.size })));
              }

              // Redirigir a la carpeta correspondiente para que el usuario vea sus fotos recién guardadas
              if (photoCount > 0 && videoCount === 0 && appCount === 0 && audioCount === 0) {
                setActiveCategory('photo');
                setActiveTab('photo');
              } else if (videoCount > 0 && photoCount === 0 && appCount === 0 && audioCount === 0) {
                setActiveCategory('video');
                setActiveTab('video');
              } else if (audioCount > 0 && photoCount === 0 && videoCount === 0 && appCount === 0) {
                setActiveCategory('audio');
                setActiveTab('audio');
              } else if (appCount > 0 && photoCount === 0 && videoCount === 0 && audioCount === 0) {
                setActiveCategory('app');
                setActiveTab('app');
              } else {
                setActiveCategory(null);
                setActiveTab('all');
              }
              setActiveSubfolder(null);
            }
          }}
          showToast={showToast}
        />
      ) : activeNavTab === 'security' ? (
        /* Screen 3: Seguridad Tab */
        <SecurityVaultView
          onOpenSettings={() => navigateToTab('settings')}
          showToast={showToast}
          sessionPin={sessionPin}
          onLock={onLock}
        />
      ) : activeNavTab === 'settings' ? (
        /* Screen 4: Configuración Tab */
        <SettingsVaultView
          masterKey={masterKey}
          sessionPin={sessionPin}
          onMasterKeyUpdated={onMasterKeyUpdated || (() => {})}
          autoLockMinutes={autoLockMinutes}
          onAutoLockChange={onAutoLockChange || (() => {})}
          lockOnTabSwitch={lockOnTabSwitch}
          onLockOnTabSwitchChange={onLockOnTabSwitchChange || (() => {})}
          onPhotosUpdated={loadVaultItems}
          onResetVault={onResetVault || (() => {})}
          showToast={showToast}
        />
      ) : activeCategory === null ? (
        /* Screen 1: Bóveda Tab Home (Exact Image 1) */
        <div className="max-w-md mx-auto w-full px-4 pb-28 flex-1 flex flex-col animate-in fade-in duration-200">
          {/* Fixed Pinned Header */}
          <header className="fixed top-0 left-0 right-0 z-30 bg-[#090d16]/95 backdrop-blur-xl border-b border-slate-800/80 safe-top-sticky pb-3 px-4 shadow-md shadow-black/40 transition-all">
            <div className="max-w-md mx-auto w-full flex items-center justify-between">
              <div>
                <h1 className="text-2xl sm:text-3xl font-black tracking-tight text-white mb-0.5">
                  Mi <span className="text-[#00e5ff]">Bóveda</span>
                </h1>
                <p className="text-[10px] font-semibold text-slate-400 tracking-[0.2em] uppercase">
                  TUS ARCHIVOS, SIEMPRE PROTEGIDOS
                </p>
              </div>
            </div>
          </header>

          {/* Spacer so content starts right below fixed header */}
          <div className="safe-top-header-spacer shrink-0 w-full mb-3" />

          {/* Recordatorio de Copia de Seguridad (> 7 días) */}
          {backupReminder.show && (
            <div className="bg-gradient-to-r from-amber-950/40 via-amber-900/20 to-amber-950/40 border border-amber-500/40 rounded-3xl p-3.5 mb-3 shadow-lg flex items-center justify-between gap-3 animate-in fade-in select-none">
              <div
                onClick={() => {
                  sessionStorage.setItem('open_backup_tab_on_settings', 'true');
                  window.dispatchEvent(new Event('nullvault_switch_settings_backup'));
                  navigateToTab('settings');
                }}
                className="flex items-center gap-3 min-w-0 cursor-pointer flex-1 group"
              >
                <div className="w-11 h-11 rounded-2xl bg-amber-500/20 border border-amber-500/30 text-amber-400 flex items-center justify-center shrink-0 group-hover:scale-105 transition-transform">
                  <Clock className="w-5 h-5" />
                </div>
                <div className="min-w-0">
                  <div className="flex items-center gap-1.5 text-amber-400 text-[10px] font-bold tracking-wider uppercase">
                    <AlertTriangle className="w-3.5 h-3.5 shrink-0" />
                    <span>Recordatorio de Respaldo</span>
                    <span className="px-1.5 py-0.2 rounded-full bg-amber-500/20 text-amber-300 text-[9px] font-mono">
                      {backupReminder.hasNeverBackedUp ? 'Pendiente' : `+${backupReminder.days} días`}
                    </span>
                  </div>
                  <h4 className="text-xs font-bold text-white truncate mt-0.5">
                    {backupReminder.hasNeverBackedUp
                      ? 'Aún no has creado una copia de seguridad'
                      : `Hace más de una semana sin respaldar (${backupReminder.days} días)`}
                  </h4>
                  <p className="text-[10px] text-amber-300 underline font-medium">
                    Toca para respaldar tus archivos ahora →
                  </p>
                </div>
              </div>
              <button
                type="button"
                onClick={(e) => {
                  e.stopPropagation();
                  dismissBackupReminder();
                }}
                className="p-1.5 text-slate-400 hover:text-white rounded-lg hover:bg-slate-800/60 shrink-0 cursor-pointer"
                title="Posponer recordatorio"
              >
                <X className="w-4 h-4" />
              </button>
            </div>
          )}

          {/* Card: BÓVEDA SELLADA (Real Data) - Compact sleek card */}
          <div className="bg-[#0a121d] border border-slate-800/90 rounded-2xl p-2.5 sm:p-3 mb-2.5 shadow-xl flex items-center justify-between">
            {/* Circular Progress Gauge */}
            <div className="relative w-11 h-11 shrink-0 flex items-center justify-center">
              <svg className="w-full h-full -rotate-90" viewBox="0 0 36 36">
                <path
                  className="text-slate-800"
                  strokeWidth="3.5"
                  stroke="currentColor"
                  fill="none"
                  d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831"
                />
                <path
                  className="text-[#00e5ff] transition-all duration-500"
                  strokeDasharray={`${storagePercent}, 100`}
                  strokeLinecap="round"
                  strokeWidth="3.5"
                  stroke="currentColor"
                  fill="none"
                  d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831"
                />
              </svg>
              <span className="absolute text-[11px] font-bold text-white font-mono">
                {storagePercent}%
              </span>
            </div>

            {/* Info */}
            <div className="flex-1 min-w-0 px-2.5">
              <p className="text-[9px] font-bold text-[#00e5ff] tracking-widest uppercase">
                BÓVEDA SELLADA
              </p>
              <h3 className="text-xs sm:text-sm font-bold text-white truncate">
                {items.length} {items.length === 1 ? 'archivo cifrado' : 'archivos cifrados'}
              </h3>
              <p className="text-[10px] text-slate-400 truncate">
                {items.length > 0
                  ? `${formatSize(totalStorageBytes)} usados · Hoy ${lastAccessTime}`
                  : `Último acceso: hoy ${lastAccessTime}`}
              </p>
            </div>

            {/* Shield with lock icon */}
            <div className="w-8 h-8 rounded-xl bg-[#00e5ff]/10 border border-[#00e5ff]/30 text-[#00e5ff] flex items-center justify-center shrink-0 shadow-md shadow-cyan-500/10">
              <Shield className="w-3.5 h-3.5" />
            </div>
          </div>

          {/* 4 Category Cards in 2x2 Grid (Compact & Sleek) */}
          <div className="grid grid-cols-2 gap-2 mb-3">
            {/* Fotos */}
            <button
              type="button"
              onClick={() => openCategory('photo')}
              className="bg-[#0a121d] border border-slate-800/80 hover:border-[#00e5ff]/40 rounded-xl p-2.5 flex flex-col justify-between text-left transition-all active:scale-[0.98] group shadow-md cursor-pointer"
            >
              <div className="flex items-center justify-between mb-1.5">
                <ImageIcon className="w-5 h-5 text-[#00e5ff]" />
                <ChevronRight className="w-3.5 h-3.5 text-slate-500 group-hover:text-[#00e5ff] transition-colors" />
              </div>
              <div>
                <p className="text-[10px] text-slate-400 font-medium">Fotos</p>
                <h4 className="text-lg font-bold text-white mt-0.5 leading-tight">
                  {counts.photos}
                </h4>
              </div>
            </button>

            {/* Videos */}
            <button
              type="button"
              onClick={() => openCategory('video')}
              className="bg-[#0a121d] border border-slate-800/80 hover:border-[#00e5ff]/40 rounded-xl p-2.5 flex flex-col justify-between text-left transition-all active:scale-[0.98] group shadow-md cursor-pointer"
            >
              <div className="flex items-center justify-between mb-1.5">
                <Video className="w-5 h-5 text-[#00e5ff]" />
                <ChevronRight className="w-3.5 h-3.5 text-slate-500 group-hover:text-[#00e5ff] transition-colors" />
              </div>
              <div>
                <p className="text-[10px] text-slate-400 font-medium">Videos</p>
                <h4 className="text-lg font-bold text-white mt-0.5 leading-tight">
                  {counts.videos}
                </h4>
              </div>
            </button>

            {/* Audio */}
            <button
              type="button"
              onClick={() => openCategory('audio')}
              className="bg-[#0a121d] border border-slate-800/80 hover:border-[#00e5ff]/40 rounded-xl p-2.5 flex flex-col justify-between text-left transition-all active:scale-[0.98] group shadow-md cursor-pointer"
            >
              <div className="flex items-center justify-between mb-1.5">
                <Music className="w-5 h-5 text-[#00e5ff]" />
                <ChevronRight className="w-3.5 h-3.5 text-slate-500 group-hover:text-[#00e5ff] transition-colors" />
              </div>
              <div>
                <p className="text-[10px] text-slate-400 font-medium">Audio</p>
                <h4 className="text-lg font-bold text-white mt-0.5 leading-tight">
                  {counts.audios}
                </h4>
              </div>
            </button>

            {/* Apps */}
            <button
              type="button"
              onClick={() => openCategory('app')}
              className="bg-[#0a121d] border border-slate-800/80 hover:border-[#00e5ff]/40 rounded-xl p-2.5 flex flex-col justify-between text-left transition-all active:scale-[0.98] group shadow-md cursor-pointer"
            >
              <div className="flex items-center justify-between mb-1.5">
                <LayoutGrid className="w-5 h-5 text-[#00e5ff]" />
                <ChevronRight className="w-3.5 h-3.5 text-slate-500 group-hover:text-[#00e5ff] transition-colors" />
              </div>
              <div>
                <p className="text-[10px] text-slate-400 font-medium">Apps</p>
                <h4 className="text-lg font-bold text-white mt-0.5 leading-tight">
                  {counts.apps}
                </h4>
              </div>
            </button>
          </div>

          {/* Section: ARCHIVOS RECIENTES (Real User Files) */}
          <div className="mb-4">
            <div className="flex items-center justify-between px-1 mb-2">
              <h3 className="text-xs font-bold text-slate-400 tracking-wider uppercase">
                ARCHIVOS RECIENTES
              </h3>
              {recentList.length > 0 && (
                <button
                  type="button"
                  onClick={() => openCategory('photo')}
                  className="text-xs font-bold text-[#00e5ff] hover:underline cursor-pointer"
                >
                  VER TODO
                </button>
              )}
            </div>

            {recentList.length > 0 ? (
              <div className="space-y-1.5">
                {recentList.map((item) => (
                  <div
                    key={item.id}
                    onClick={() => {
                      if (item.realItem) {
                        setActiveViewerItem(item.realItem);
                      }
                    }}
                    className="flex items-center justify-between p-2 rounded-xl bg-[#0a121d]/70 hover:bg-[#0a121d] border border-slate-800/70 hover:border-slate-700 cursor-pointer transition-all active:scale-[0.99] group shadow-sm"
                  >
                    <div className="flex items-center gap-2.5 min-w-0">
                      <div className="w-10 h-10 rounded-lg overflow-hidden relative bg-slate-900 border border-slate-800 shrink-0">
                        {item.previewUrl ? (
                          <img
                            src={item.previewUrl}
                            alt={item.name}
                            className="w-full h-full object-cover group-hover:scale-105 transition-transform"
                          />
                        ) : (
                          <div className="w-full h-full flex items-center justify-center bg-slate-800 text-[#00e5ff]">
                            {item.category === 'audio' ? (
                              <Music className="w-4 h-4" />
                            ) : item.category === 'app' ? (
                              <LayoutGrid className="w-4 h-4" />
                            ) : (
                              <FileText className="w-4 h-4" />
                            )}
                          </div>
                        )}
                        <div className="absolute inset-0 flex items-center justify-center bg-black/40">
                          <Lock className="w-3 h-3 text-white drop-shadow" />
                        </div>
                      </div>

                      <div className="min-w-0">
                        <h4 className="text-xs font-semibold text-white truncate group-hover:text-[#00e5ff] transition-colors">
                          {item.name}
                        </h4>
                        <p className="text-[10px] text-slate-400">
                          {item.size}
                        </p>
                      </div>
                    </div>

                    <div className="flex items-center gap-1.5 shrink-0 ml-2">
                      <span className="text-[10px] text-slate-400 font-mono">
                        {item.time}
                      </span>
                      <ChevronRight className="w-3.5 h-3.5 text-slate-600 group-hover:text-slate-400 transition-colors" />
                    </div>
                  </div>
                ))}
              </div>
            ) : (
              <div className="py-6 px-4 rounded-2xl bg-[#0a121d]/60 border border-dashed border-slate-800 text-center flex flex-col items-center justify-center">
                <div className="w-10 h-10 rounded-xl bg-slate-800/80 text-slate-400 flex items-center justify-center mb-2">
                  <Clock className="w-4 h-4 text-[#00e5ff]" />
                </div>
                <h4 className="text-xs font-bold text-white">No hay archivos recientes</h4>
                <p className="text-[11px] text-slate-400 max-w-xs mt-0.5">
                  Guarda fotos, videos, audios o documentos para verlos cifrados aquí.
                </p>
                <button
                  type="button"
                  onClick={() => navigateToTab('import')}
                  className="mt-3 px-3.5 py-1.5 rounded-full bg-[#00e5ff] hover:bg-[#00cce6] text-black text-xs font-bold flex items-center gap-1.5 transition-all active:scale-95 shadow-md shadow-cyan-500/20 cursor-pointer"
                >
                  <Plus className="w-3.5 h-3.5 stroke-[3]" />
                  <span>Importar a Bóveda</span>
                </button>
              </div>
            )}
          </div>
        </div>
      ) : (
        /* Detailed Category View */
        <div className="max-w-4xl w-full mx-auto p-4 flex-1 flex flex-col animate-in fade-in duration-200">
          {/* Sticky Header & Filter Controls Container */}
          <header className="sticky top-0 z-30 bg-[#090d16]/95 backdrop-blur-xl border-b border-slate-800/80 px-2 safe-top-sticky pb-2 mb-3 shadow-xl -mx-4 sm:mx-0 sm:px-2">
            <div className="flex items-center justify-between gap-3 mb-2.5">
              <button
                type="button"
                onClick={() => {
                  if (activeSubfolder !== null) {
                    setActiveSubfolder(null);
                  } else {
                    setActiveCategory(null);
                  }
                }}
                className="p-1 -ml-1 rounded-xl text-slate-300 hover:text-white hover:bg-slate-800/60 transition-colors flex items-center gap-2 active:scale-95 cursor-pointer"
              >
                <ArrowLeft className="w-5 h-5 text-[#00e5ff]" />
                <div className="text-left">
                  <div className="flex items-center gap-2">
                    <h1 className="font-bold text-base sm:text-lg text-white">
                      {activeCategory === 'photo'
                        ? 'Fotos'
                        : activeCategory === 'video'
                        ? 'Videos'
                        : activeCategory === 'audio'
                        ? 'Audio'
                        : 'Apps'}
                      {activeSubfolder !== null && (
                        <span className="text-[#00e5ff] font-extrabold ml-1.5">
                          / {activeSubfolder}
                        </span>
                      )}
                    </h1>
                    <span className="text-xs text-slate-400 font-mono">
                      (
                      {activeSubfolder !== null
                        ? filteredItems.length
                        : activeCategory === 'photo'
                        ? counts.photos
                        : activeCategory === 'video'
                        ? counts.videos
                        : activeCategory === 'audio'
                        ? counts.audios
                        : counts.apps}
                      )
                    </span>
                  </div>
                  <p className="text-[10px] text-slate-400">
                    {activeSubfolder !== null ? 'Volver a Carpetas' : 'Volver a Bóveda'}
                  </p>
                </div>
              </button>

              <div className="flex items-center gap-1.5">
                <button
                  onClick={() => navigateToTab('settings')}
                  className="p-2 rounded-xl bg-slate-800/60 hover:bg-slate-700/80 text-slate-300 hover:text-white transition-colors cursor-pointer"
                  title="Ajustes"
                >
                  <Settings className="w-4 h-4" />
                </button>
                <button
                  onClick={onLock}
                  className="p-2 rounded-xl bg-rose-500/10 hover:bg-rose-500/20 text-rose-400 border border-rose-500/20 transition-colors cursor-pointer"
                  title="Bloquear"
                >
                  <Lock className="w-4 h-4" />
                </button>
              </div>
            </div>

            {/* Search & Layout Bar */}
            <div className="flex items-center gap-2 mb-2.5">
              <div className="relative flex-1">
                <Search className="w-4 h-4 text-slate-500 absolute left-3.5 top-1/2 -translate-y-1/2" />
                <input
                  type="text"
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  placeholder="Buscar archivos..."
                  className="w-full bg-[#0a121d] border border-slate-800 rounded-2xl pl-10 pr-4 py-2 text-xs text-slate-200 placeholder-slate-500 focus:outline-none focus:border-[#00e5ff] transition-colors"
                />
              </div>

              <div className="flex items-center gap-1 bg-[#0a121d] border border-slate-800 rounded-2xl p-0.5">
                <button
                  type="button"
                  onClick={toggleAspectMode}
                  className={`px-2 py-1.5 rounded-xl transition-all flex items-center gap-1 cursor-pointer text-[10px] font-semibold ${
                    aspectMode === 'real'
                      ? 'bg-[#00e5ff] text-black shadow-sm'
                      : 'text-slate-400 hover:text-white'
                  }`}
                  title={aspectMode === 'real' ? 'Proporción y tamaño real (sin recorte)' : 'Cuadrícula cuadrada 1:1 (recortada)'}
                >
                  <Maximize2 className="w-3.5 h-3.5" />
                  <span className="hidden sm:inline">{aspectMode === 'real' ? 'Tamaño real' : '1:1'}</span>
                </button>
                <div className="w-[1px] h-3.5 bg-slate-800 mx-0.5" />
                <button
                  onClick={() => setGridColumns(2)}
                  className={`p-1.5 rounded-xl transition-colors cursor-pointer ${
                    gridColumns === 2 ? 'bg-[#00e5ff] text-black font-bold' : 'text-slate-400 hover:text-white'
                  }`}
                  title="2 Columnas"
                >
                  <LayoutGrid className="w-3.5 h-3.5" />
                </button>
                <button
                  onClick={() => setGridColumns(3)}
                  className={`p-1.5 rounded-xl transition-colors cursor-pointer ${
                    gridColumns === 3 ? 'bg-[#00e5ff] text-black font-bold' : 'text-slate-400 hover:text-white'
                  }`}
                  title="3 Columnas"
                >
                  <Grid3X3 className="w-3.5 h-3.5" />
                </button>
              </div>
            </div>

            {/* Content Type Filter Tabs */}
            <div className="flex items-center gap-1.5 overflow-x-auto pb-1 mb-1.5 no-scrollbar text-xs w-full max-w-full">
              <button
                type="button"
                onClick={() => {
                  setActiveCategory('photo');
                  setActiveTab('photo');
                  setActiveSubfolder(null);
                }}
                className={`px-3 py-1.5 rounded-xl font-medium whitespace-nowrap transition-colors flex items-center gap-1.5 shrink-0 cursor-pointer ${
                  (activeCategory === 'photo' || activeTab === 'photo') && activeTab !== 'favorites'
                    ? 'bg-[#00e5ff] text-black font-bold shadow-md shadow-cyan-500/25'
                    : 'bg-[#131c2e] text-slate-400 hover:text-slate-200 border border-slate-800'
                }`}
              >
                <ImageIcon className="w-3.5 h-3.5" />
                <span>Fotos ({counts.photos})</span>
              </button>

              <button
                type="button"
                onClick={() => {
                  setActiveCategory('video');
                  setActiveTab('video');
                  setActiveSubfolder(null);
                }}
                className={`px-3 py-1.5 rounded-xl font-medium whitespace-nowrap transition-colors flex items-center gap-1.5 shrink-0 cursor-pointer ${
                  activeCategory === 'video' || activeTab === 'video'
                    ? 'bg-[#00e5ff] text-black font-bold shadow-md shadow-cyan-500/25'
                    : 'bg-[#131c2e] text-slate-400 hover:text-slate-200 border border-slate-800'
                }`}
              >
                <Film className="w-3.5 h-3.5" />
                <span>Videos ({counts.videos})</span>
              </button>

              <button
                type="button"
                onClick={() => {
                  setActiveCategory('audio');
                  setActiveTab('audio');
                  setActiveSubfolder(null);
                }}
                className={`px-3 py-1.5 rounded-xl font-medium whitespace-nowrap transition-colors flex items-center gap-1.5 shrink-0 cursor-pointer ${
                  activeCategory === 'audio' || activeTab === 'audio'
                    ? 'bg-[#00e5ff] text-black font-bold shadow-md shadow-cyan-500/25'
                    : 'bg-[#131c2e] text-slate-400 hover:text-slate-200 border border-slate-800'
                }`}
              >
                <Music className="w-3.5 h-3.5" />
                <span>Audio ({counts.audios})</span>
              </button>

              <button
                type="button"
                onClick={() => {
                  setActiveCategory('app');
                  setActiveTab('app');
                  setActiveSubfolder(null);
                }}
                className={`px-3 py-1.5 rounded-xl font-medium whitespace-nowrap transition-colors flex items-center gap-1.5 shrink-0 cursor-pointer ${
                  activeCategory === 'app' || activeTab === 'app'
                    ? 'bg-[#00e5ff] text-black font-bold shadow-md shadow-cyan-500/25'
                    : 'bg-[#131c2e] text-slate-400 hover:text-slate-200 border border-slate-800'
                }`}
              >
                <Folder className="w-3.5 h-3.5" />
                <span>Apps ({counts.apps})</span>
              </button>

              <button
                type="button"
                onClick={() => {
                  setActiveTab('favorites');
                  setActiveSubfolder(null);
                }}
                className={`px-3 py-1.5 rounded-xl font-medium whitespace-nowrap transition-colors flex items-center gap-1.5 shrink-0 cursor-pointer ${
                  activeTab === 'favorites'
                    ? 'bg-rose-600 text-white font-bold shadow-md shadow-rose-600/25'
                    : 'bg-[#131c2e] text-slate-400 hover:text-slate-200 border border-slate-800'
                }`}
              >
                <Heart className="w-3.5 h-3.5" />
                <span>Favoritos ({counts.favs})</span>
              </button>
            </div>
          </header>

        {/* Items Grid & Carpetas View */}
        {isLoading ? (
          <div className="py-20 flex flex-col items-center justify-center text-slate-500">
            <Cpu className="w-8 h-8 animate-spin text-blue-500 mb-3" />
            <p className="text-xs">Descifrando datos protegidos de Caja fuerte...</p>
          </div>
        ) : (
          /* VISTA DE CONTENIDO DE LA SECCIÓN */
          <div className="space-y-4">
            {/* Si NO hay carpeta seleccionada (activeSubfolder === null), MOSTRAR LAS CARPETAS DE ESTA SECCIÓN */}
            {activeSubfolder === null && !searchQuery && (
              <div className="mb-4">
                <div className="flex items-center justify-between mb-2 px-1">
                  <h3 className="text-xs font-bold text-slate-400 tracking-wider uppercase flex items-center gap-1.5">
                    <Folder className="w-3.5 h-3.5 text-[#00e5ff]" />
                    <span>Carpetas de {activeTab === 'photo' ? 'Fotos' : activeTab === 'video' ? 'Videos' : activeTab === 'audio' ? 'Audio' : activeTab === 'app' ? 'Apps' : 'Favoritos'}</span>
                  </h3>
                  <button
                    type="button"
                    onClick={() => setIsCreateFolderOpen(true)}
                    className="text-xs font-bold text-[#00e5ff] hover:underline flex items-center gap-1 cursor-pointer"
                  >
                    <FolderPlus className="w-3.5 h-3.5" />
                    <span>+ Nueva Carpeta</span>
                  </button>
                </div>

                {/* Lista Vertical de Carpetas de la Sección */}
                <div className="flex flex-col divide-y divide-slate-800/60 bg-[#0a121d]/60 rounded-2xl border border-slate-800/80 overflow-hidden mb-5">
                  {activeCategoryFolders.map((fName) => {
                    const folderItems = items.filter((it) => {
                      const matchesCat =
                        activeTab === 'favorites'
                          ? it.record.isFavorite
                          : activeTab === 'all'
                          ? true
                          : it.record.category === activeTab;
                      const itemAlbum = it.record.album || 'General';
                      const matchesFolder = itemAlbum === fName;
                      return matchesCat && matchesFolder;
                    });
                    const count = folderItems.length;

                    return (
                      <div
                        key={fName}
                        onClick={() => setActiveSubfolder(fName)}
                        className="flex items-center justify-between p-3.5 hover:bg-slate-800/50 cursor-pointer transition-colors group"
                      >
                        <div className="flex items-center gap-3 min-w-0">
                          <div className="w-9 h-9 rounded-xl bg-indigo-500/10 text-[#00e5ff] flex items-center justify-center shrink-0 group-hover:scale-105 transition-transform">
                            <Folder className="w-5 h-5 text-[#00e5ff]" />
                          </div>
                          <div className="min-w-0">
                            <h4 className="text-sm font-bold text-slate-200 truncate group-hover:text-[#00e5ff] transition-colors">
                              {fName}
                            </h4>
                            <p className="text-[11px] text-slate-400 font-mono">
                              {count} {count === 1 ? 'archivo' : 'archivos'}
                            </p>
                          </div>
                        </div>

                        <div className="flex items-center gap-2 shrink-0">
                          {fName !== 'General' && (
                            <div
                              onClick={(e) => e.stopPropagation()}
                              className="flex items-center gap-1 opacity-70 group-hover:opacity-100 transition-opacity"
                            >
                              <button
                                type="button"
                                onClick={() => setFolderToRename(fName)}
                                className="p-1.5 rounded-lg text-slate-400 hover:text-white hover:bg-slate-800"
                                title="Renombrar carpeta"
                              >
                                <Edit3 className="w-3.5 h-3.5" />
                              </button>
                              <button
                                type="button"
                                onClick={() => handleDeleteFolder(fName)}
                                className="p-1.5 rounded-lg text-slate-400 hover:text-rose-400 hover:bg-rose-500/10"
                                title="Eliminar carpeta"
                              >
                                <Trash2 className="w-3.5 h-3.5" />
                              </button>
                            </div>
                          )}
                          <ChevronRight className="w-4 h-4 text-slate-500 group-hover:text-[#00e5ff] group-hover:translate-x-0.5 transition-all" />
                        </div>
                      </div>
                    );
                  })}
                </div>
              </div>
            )}

            {/* Grid de Archivos Cifrados - Solo se muestra al abrir una carpeta seleccionada o buscar */}
            {(activeSubfolder !== null || searchQuery) && (
              <div className="space-y-3">
                {activeSubfolder !== null && !searchQuery && (
                  <div className="flex items-center justify-between px-1 mb-2">
                    <button
                      type="button"
                      onClick={() => setActiveSubfolder(null)}
                      className="text-xs font-bold text-[#00e5ff] hover:underline flex items-center gap-1.5 cursor-pointer"
                    >
                      <ArrowLeft className="w-3.5 h-3.5" />
                      <span>Volver a las Carpetas</span>
                    </button>
                    <span className="text-xs text-slate-400 font-mono">
                      {filteredItems.length} {filteredItems.length === 1 ? 'archivo' : 'archivos'}
                    </span>
                  </div>
                )}

                {filteredItems.length === 0 ? (
          <div className="text-center py-16 px-4 bg-[#131c2e]/40 border border-dashed border-slate-800 rounded-3xl">
            <div className="w-16 h-16 bg-slate-800/60 rounded-2xl flex items-center justify-center mx-auto mb-4 text-slate-500">
              {activeTab === 'photo' ? (
                <ImageIcon className="w-8 h-8 text-blue-400" />
              ) : activeTab === 'audio' ? (
                <Music className="w-8 h-8 text-amber-400" />
              ) : activeTab === 'video' ? (
                <Film className="w-8 h-8 text-purple-400" />
              ) : activeTab === 'app' ? (
                <Folder className="w-8 h-8 text-emerald-400" />
              ) : (
                <FolderArchive className="w-8 h-8 text-blue-400" />
              )}
            </div>
            <h3 className="text-base font-semibold text-slate-200 mb-1">
              {searchQuery
                ? 'No se encontraron resultados'
                : activeTab === 'photo'
                ? 'No hay imágenes en Caja fuerte'
                : activeTab === 'audio'
                ? 'No hay audios en Caja fuerte'
                : activeTab === 'video'
                ? 'No hay videos en Caja fuerte'
                : activeTab === 'app'
                ? 'No hay archivos en Caja fuerte'
                : 'Sin favoritos aún'}
            </h3>
            <p className="text-xs text-slate-400 max-w-sm mx-auto mb-6">
              {activeTab === 'photo'
                ? 'Añade fotos para mantenerlas cifradas de forma privada en tu teléfono.'
                : activeTab === 'audio'
                ? 'Añade audios o notas de voz para protegerlos en tu Caja fuerte.'
                : activeTab === 'video'
                ? 'Guarda tus videos privados para reproducirlos de forma segura en tu teléfono.'
                : activeTab === 'app'
                ? 'Guarda aplicaciones (.apk) o documentos cifrados localmente.'
                : 'Toca el corazón en cualquier elemento para destacarlo aquí.'}
            </p>
            <div className="flex flex-wrap items-center justify-center gap-2.5">
              {(activeTab === 'photo' || activeTab === 'all') && (
                <button
                  type="button"
                  onClick={() => mediaInputRef.current?.click()}
                  className="px-4 py-2.5 bg-blue-600 hover:bg-blue-500 text-white text-xs font-semibold rounded-xl shadow-lg shadow-blue-600/30 transition-all flex items-center gap-2 active:scale-95"
                >
                  <ImagePlus className="w-4 h-4" />
                  <span>Añadir Fotos</span>
                </button>
              )}
              {activeTab === 'audio' && (
                <button
                  type="button"
                  onClick={() => audioInputRef.current?.click()}
                  className="px-4 py-2.5 bg-amber-600 hover:bg-amber-500 text-white text-xs font-semibold rounded-xl shadow-lg shadow-amber-600/30 transition-all flex items-center gap-2 active:scale-95"
                >
                  <Music className="w-4 h-4" />
                  <span>Añadir Audio</span>
                </button>
              )}
              {activeTab === 'video' && (
                <button
                  type="button"
                  onClick={() => mediaInputRef.current?.click()}
                  className="px-4 py-2.5 bg-purple-600 hover:bg-purple-500 text-white text-xs font-semibold rounded-xl shadow-lg shadow-purple-600/30 transition-all flex items-center gap-2 active:scale-95"
                >
                  <Film className="w-4 h-4" />
                  <span>Añadir Videos</span>
                </button>
              )}
              {activeTab === 'app' && (
                <button
                  type="button"
                  onClick={() => appInputRef.current?.click()}
                  className="px-4 py-2.5 bg-emerald-600 hover:bg-emerald-500 text-white text-xs font-semibold rounded-xl shadow-lg shadow-emerald-600/30 transition-all flex items-center gap-2 active:scale-95"
                >
                  <Folder className="w-4 h-4" />
                  <span>Añadir Archivo (.apk)</span>
                </button>
              )}
              <button
                type="button"
                onClick={() => setIsCameraOpen(true)}
                className="px-4 py-2.5 bg-[#131c2e] hover:bg-slate-800 text-slate-300 text-xs font-semibold rounded-xl border border-slate-700 transition-all flex items-center gap-2 active:scale-95"
              >
                <Camera className="w-4 h-4 text-blue-400" />
                <span>Usar Cámara</span>
              </button>
            </div>
          </div>
        ) : (
          <div
            className={`grid gap-2.5 sm:gap-3 ${
              gridColumns === 2
                ? 'grid-cols-2'
                : gridColumns === 3
                ? 'grid-cols-3'
                : 'grid-cols-4'
            }`}
          >
            {filteredItems.map((item) => {
              const isSelected = selectedIds.has(item.id);
              const isPhoto = item.record.category === 'photo';
              const isVideo = item.record.category === 'video';
              const isAudio = item.record.category === 'audio';
              const isApp = item.record.category === 'app';

              return (
                <div
                  key={item.id}
                  onPointerDown={() => handleCardPointerDown(item.id)}
                  onPointerUp={handleCardPointerUp}
                  onPointerCancel={handleCardPointerUp}
                  onClick={() => {
                    if (isLongPressTriggered.current) {
                      isLongPressTriggered.current = false;
                      return;
                    }
                    if (isSelectMode) {
                      toggleSelectItem(item.id, {} as React.MouseEvent);
                    } else {
                      setActiveViewerItem(item);
                    }
                  }}
                  className={`group relative aspect-square rounded-2xl overflow-hidden bg-slate-900 border cursor-pointer transition-all duration-200 select-none ${
                    isSelected
                      ? 'border-blue-500 ring-2 ring-blue-500/50 scale-[0.98]'
                      : 'border-slate-800/80 hover:border-slate-700 active:scale-95'
                  }`}
                >
                  {/* Photo Display */}
                  {isPhoto && (
                    <img
                      src={item.blobUrl}
                      alt={item.record.name}
                      className={`w-full h-full ${
                        aspectMode === 'real' ? 'object-contain p-0.5' : 'object-cover'
                      } transition-transform duration-300 group-hover:scale-105 pointer-events-none`}
                      loading="lazy"
                    />
                  )}

                  {/* Video Thumbnail Display */}
                  {isVideo && (
                    <div className="w-full h-full bg-slate-950 flex flex-col items-center justify-center relative pointer-events-none">
                      <video
                        src={item.blobUrl}
                        preload="metadata"
                        className={`w-full h-full ${
                          aspectMode === 'real' ? 'object-contain p-0.5' : 'object-cover'
                        } opacity-75 group-hover:opacity-95 transition-opacity`}
                      />
                      <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
                        <div className="w-10 h-10 rounded-full bg-black/60 backdrop-blur-md border border-white/20 flex items-center justify-center text-white group-hover:scale-110 transition-transform">
                          <Play className="w-4 h-4 fill-white ml-0.5" />
                        </div>
                      </div>
                      <div className="absolute top-2 left-2 px-1.5 py-0.5 rounded bg-black/60 backdrop-blur-md text-[10px] font-mono text-purple-300 flex items-center gap-1">
                        <Film className="w-3 h-3" />
                        <span>VIDEO</span>
                      </div>
                    </div>
                  )}

                  {/* Audio Card Display */}
                  {isAudio && (
                    <div className="w-full h-full bg-gradient-to-br from-[#1d170e] to-slate-950 p-3.5 flex flex-col justify-between items-center text-center pointer-events-none">
                      <div className="w-full flex items-center justify-between text-[10px] text-amber-400">
                        <span className="font-mono font-bold bg-amber-500/10 px-1.5 py-0.5 rounded border border-amber-500/20">
                          {item.record.name.split('.').pop()?.toUpperCase() || 'AUDIO'}
                        </span>
                        <span className="text-slate-400">{formatSize(item.record.size)}</span>
                      </div>

                      <div className="w-12 h-12 rounded-2xl bg-amber-500/15 border border-amber-500/30 flex items-center justify-center text-amber-400 group-hover:scale-110 transition-transform shadow-md shadow-amber-500/10">
                        <Music className="w-6 h-6 text-amber-400" />
                      </div>

                      <div className="w-full text-center">
                        <p className="text-xs font-semibold text-slate-200 truncate w-full">
                          {item.record.name}
                        </p>
                        <p className="text-[10px] text-amber-400/80">Audio seguro</p>
                      </div>
                    </div>
                  )}

                  {/* App / File Card Display */}
                  {isApp && (
                    <div className="w-full h-full bg-gradient-to-br from-[#131c2e] to-slate-950 p-3.5 flex flex-col justify-between items-center text-center pointer-events-none">
                      <div className="w-full flex items-center justify-between text-[10px] text-emerald-400">
                        <span className="font-mono font-bold bg-emerald-500/10 px-1.5 py-0.5 rounded border border-emerald-500/20">
                          {item.record.name.split('.').pop()?.toUpperCase() || 'FILE'}
                        </span>
                        <span className="text-slate-400">{formatSize(item.record.size)}</span>
                      </div>

                      <div className="w-12 h-12 rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center text-emerald-400 group-hover:scale-110 transition-transform">
                        {item.record.name.endsWith('.apk') ? (
                          <Smartphone className="w-6 h-6" />
                        ) : (
                          <Folder className="w-6 h-6" />
                        )}
                      </div>

                      <div className="w-full text-center">
                        <p className="text-xs font-semibold text-slate-200 truncate w-full">
                          {item.record.name}
                        </p>
                        <p className="text-[10px] text-slate-500">Cifrado local</p>
                      </div>
                    </div>
                  )}

                  {/* Gradient overlay for photos/videos */}
                  {(isPhoto || isVideo) && (
                    <div className="absolute inset-0 bg-gradient-to-t from-black/70 via-transparent to-black/20 opacity-0 group-hover:opacity-100 transition-opacity pointer-events-none" />
                  )}

                  {/* Top-right Favorite indicator */}
                  {item.record.isFavorite && (
                    <div className="absolute top-2 right-2 p-1 bg-black/60 backdrop-blur-md rounded-lg text-rose-500 z-10 pointer-events-none">
                      <Heart className="w-3.5 h-3.5 fill-rose-500" />
                    </div>
                  )}

                  {/* Selection indicator - circular cyan checkmark */}
                  {isSelectMode && (
                    <div className="absolute top-2 left-2 z-20 pointer-events-none">
                      <div
                        className={`w-6 h-6 rounded-full flex items-center justify-center transition-all ${
                          isSelected
                            ? 'bg-[#00e5ff] text-black shadow-md border-2 border-white'
                            : 'bg-black/60 border border-white/60 text-transparent'
                        }`}
                      >
                        <Check className="w-3.5 h-3.5 stroke-[3]" />
                      </div>
                    </div>
                  )}

                  {/* Bottom quick metadata */}
                  {(isPhoto || isVideo) && (
                    <div className="absolute bottom-2 left-2 right-2 flex items-center justify-between text-[10px] text-white/90 opacity-0 group-hover:opacity-100 transition-opacity drop-shadow z-10 pointer-events-none">
                      <span className="truncate max-w-[120px]">{item.record.name}</span>
                      <span>{formatSize(item.record.size)}</span>
                    </div>
                  )}
                </div>
              );
            })}
          </div>
        )}
              </div>
            )}
            </div>
          )}
        </div>
      )}

      {/* Floating Bottom Navigation Bar for Selection Mode */}
      {isSelectMode || selectedIds.size > 0 ? (
        <div className="fixed bottom-3 sm:bottom-4 left-0 right-0 z-50 px-2 sm:px-4 flex justify-center pointer-events-none">
          <nav className="pointer-events-auto w-full max-w-lg bg-[#0a121d]/95 backdrop-blur-2xl border border-slate-700/80 rounded-2xl sm:rounded-full p-1.5 sm:p-2 flex items-center justify-between shadow-2xl shadow-black/80 gap-1 sm:gap-1.5 overflow-x-auto [scrollbar-width:none] [-ms-overflow-style:none] [&::-webkit-scrollbar]:hidden touch-pan-x">
            <div className="flex items-center gap-1 shrink-0">
              <button
                type="button"
                onClick={() => {
                  setIsSelectMode(false);
                  setSelectedIds(new Set());
                }}
                className="px-2.5 py-1.5 rounded-xl bg-[#162032] border border-slate-700 text-slate-100 hover:bg-slate-700 text-xs font-semibold flex items-center gap-1 transition-all active:scale-95 cursor-pointer shrink-0"
                title="Listo / Salir de selección"
              >
                <X className="w-3.5 h-3.5 text-slate-300" />
                <span>Listo</span>
              </button>
              <button
                type="button"
                onClick={handleSelectAll}
                className="px-1.5 py-1 text-xs text-[#00e5ff] hover:underline font-bold transition-all cursor-pointer shrink-0"
              >
                {selectedIds.size === filteredItems.length ? 'Ninguno' : 'Todos'}
              </button>
            </div>

            <div className="flex items-center gap-1 sm:gap-1.5 shrink-0">
              {activeTab !== 'all' && activeTab !== 'favorites' && (
                <button
                  type="button"

                  disabled={selectedIds.size === 0}

/**
 * IndexedDB storage engine for Bóveda Segura.
 * Stores AES-GCM encrypted binary photos, videos, and application files locally.
 */

import { hexToBuf } from '../crypto/vaultCrypto';

export type VaultItemCategory = 'photo' | 'video' | 'audio' | 'app';

export function detectCategory(filename: string, mimeType: string): VaultItemCategory {
  const lowerName = filename.toLowerCase();
  const lowerMime = (mimeType || '').toLowerCase();

  if (
    lowerMime.startsWith('video/') ||
    /\.(mp4|webm|mkv|mov|avi|flv|m4v|3gp|wmv|ts)$/i.test(lowerName)
  ) {
    return 'video';
  }

  if (
    lowerMime.startsWith('audio/') ||
    /\.(mp3|wav|ogg|m4a|aac|flac|opus|wma|amr|mid|midi)$/i.test(lowerName)
  ) {
    return 'audio';
  }

  if (
    lowerMime.includes('android.package-archive') ||
    lowerMime.includes('x-msdownload') ||
    lowerMime.includes('zip') ||
    lowerMime.includes('x-zip') ||
    lowerMime.includes('octet-stream') ||
    lowerMime.includes('pdf') ||
    /\.(apk|exe|msi|dmg|pkg|deb|rpm|ipa|zip|rar|tar|gz|7z|bin|iso|appimage|jar|pdf|docx?|xlsx?|pptx?|txt)$/i.test(lowerName)
  ) {
    return 'app';
  }

  if (
    lowerMime.startsWith('image/') ||
    /\.(jpg|jpeg|png|gif|webp|bmp|svg|avif|heic|ico)$/i.test(lowerName)
  ) {
    return 'photo';
  }

  // Fallback for document or application packages
  return 'app';
}

export interface EncryptedVaultItem {
  id: string;
  timestamp: number;
  name: string;
  mimeType: string;
  size: number;
  iv: number[];
  data: ArrayBuffer;
  album?: string;
  isFavorite?: boolean;
  category: VaultItemCategory;
  lastModified?: number;
  originalUri?: string;
}

// Backwards compatibility alias
export type EncryptedPhotoRecord = EncryptedVaultItem;

const DB_NAME = 'SecurePhotoVaultDB';
const STORE_NAME = 'encrypted_photos';
const DB_VERSION = 2;

let dbInstance: IDBDatabase | null = null;

export function openVaultDatabase(): Promise<IDBDatabase> {
  if (dbInstance) return Promise.resolve(dbInstance);

  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION);

    request.onerror = () => {
      reject(request.error);
    };

    request.onsuccess = () => {
      dbInstance = request.result;
      resolve(dbInstance);
    };

    request.onupgradeneeded = (event) => {
      const db = (event.target as IDBOpenDBRequest).result;
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        const store = db.createObjectStore(STORE_NAME, { keyPath: 'id' });
        store.createIndex('timestamp', 'timestamp', { unique: false });
      }
    };
  });
}

export async function saveEncryptedItem(record: EncryptedVaultItem): Promise<void> {
  return saveEncryptedItems([record]);
}

export async function saveEncryptedItems(records: EncryptedVaultItem[]): Promise<void> {
  if (!records || records.length === 0) return;
  const db = await openVaultDatabase();
  return new Promise((resolve, reject) => {
    const transaction = db.transaction([STORE_NAME], 'readwrite');
    const store = transaction.objectStore(STORE_NAME);

    transaction.oncomplete = () => resolve();
    transaction.onerror = () => reject(transaction.error);
    transaction.onabort = () => reject(transaction.error);

    for (const record of records) {
      store.put(record);
    }
  });
}

export async function getAllEncryptedItems(): Promise<EncryptedVaultItem[]> {
  const db = await openVaultDatabase();
  return new Promise((resolve, reject) => {
    const transaction = db.transaction([STORE_NAME], 'readonly');
    const store = transaction.objectStore(STORE_NAME);
    const request = store.getAll();

    request.onsuccess = () => {
      const results: EncryptedVaultItem[] = (request.result || []).map((raw: any) => {
        const name = raw.name || `item_${raw.id}`;
        const mimeType = raw.mimeType || (raw.category === 'video' ? 'video/mp4' : 'image/jpeg');
        const category = raw.category || detectCategory(name, mimeType);

        return {
          id: raw.id,
          timestamp: raw.timestamp || Date.now(),
          name,
          mimeType,
          size: raw.size || (raw.data ? raw.data.byteLength : 0),
          iv: raw.iv,
          data: raw.data,
          album: raw.album || 'General',
          isFavorite: !!raw.isFavorite,
          category,
          lastModified: raw.lastModified,
          originalUri: raw.originalUri,
        };
      });

      resolve(results);
    };
    request.onerror = () => reject(request.error);
  });
}

export async function getEncryptedItemById(id: string): Promise<EncryptedVaultItem | null> {
  const db = await openVaultDatabase();
  return new Promise((resolve, reject) => {
    const transaction = db.transaction([STORE_NAME], 'readonly');
    const store = transaction.objectStore(STORE_NAME);
    const request = store.get(id);

    request.onsuccess = () => {
      const raw = request.result;
      if (!raw) {
        resolve(null);
        return;
      }
      const name = raw.name || `item_${raw.id}`;
      const mimeType = raw.mimeType || 'image/jpeg';
      const category = raw.category || detectCategory(name, mimeType);

      resolve({
        id: raw.id,
        timestamp: raw.timestamp,
        name,
        mimeType,
        size: raw.size || (raw.data ? raw.data.byteLength : 0),
        iv: raw.iv,
        data: raw.data,
        album: raw.album || 'General',
        isFavorite: !!raw.isFavorite,
        category,
      });
    };
    request.onerror = () => reject(request.error);
  });
}

export async function updateItemMetadata(
  id: string,
  updates: Partial<Omit<EncryptedVaultItem, 'id' | 'data' | 'iv'>>
): Promise<void> {
  const db = await openVaultDatabase();
  return new Promise((resolve, reject) => {
    const transaction = db.transaction([STORE_NAME], 'readwrite');
    const store = transaction.objectStore(STORE_NAME);
    const request = store.get(id);

    request.onsuccess = () => {
      const existing = request.result;
      if (!existing) {
        reject(new Error('Record not found'));
        return;
      }
      const updated = { ...existing, ...updates };
      const putRequest = store.put(updated);
      putRequest.onsuccess = () => resolve();
      putRequest.onerror = () => reject(putRequest.error);
    };
    request.onerror = () => reject(request.error);
  });
}

export async function batchUpdateAlbum(ids: string[], album: string): Promise<void> {
  const db = await openVaultDatabase();
  return new Promise((resolve, reject) => {
    const transaction = db.transaction([STORE_NAME], 'readwrite');
    const store = transaction.objectStore(STORE_NAME);

    ids.forEach((id) => {
      const getReq = store.get(id);
      getReq.onsuccess = () => {
        if (getReq.result) {
          store.put({ ...getReq.result, album });
        }
      };
    });

    transaction.oncomplete = () => resolve();

 * Creates exportable JSON containing encrypted blobs as Base64 strings.
 */
/**
 * Highly optimized native FileReader-based ArrayBuffer to Base64 conversion.
 * Leverages native Chrome C++ bindings for zero JS-side memory allocations.
 */
async function arrayBufferToBase64Async(buffer: ArrayBuffer | undefined | null): Promise<string> {
  if (!buffer) return '';
  return new Promise((resolve) => {
    try {
      const blob = new Blob([buffer]);
      const reader = new FileReader();
      reader.onloadend = () => {
        try {
          const dataUrl = reader.result as string;
          if (!dataUrl) {
            resolve('');
            return;
          }
          const commaIndex = dataUrl.indexOf(',');
          const base64 = commaIndex > -1 ? dataUrl.substring(commaIndex + 1) : '';
          resolve(base64);
        } catch (e) {
          console.error('Error extracting base64:', e);
          resolve('');
        }
      };
      reader.onerror = () => resolve('');
      reader.readAsDataURL(blob);
    } catch (err) {
      console.warn('Native conversion failed, returning empty:', err);
      resolve('');
    }
  });
}

/**
 * Fast native conversion from Base64 to ArrayBuffer without fetch data URL memory limits.
 * Uses chunked decoding for large files to avoid V8 mobile heap exhaustion and atob string limits.
 */
function base64ToArrayBuffer(rawBase64: string): ArrayBuffer {
  if (!rawBase64) return new ArrayBuffer(0);
  try {
    let clean = rawBase64;
    const commaIdx = clean.indexOf(',');
    if (commaIdx !== -1) {
      clean = clean.substring(commaIdx + 1);
    }
    clean = clean.replace(/[\r\n\s]/g, '');

    const padding = clean.endsWith('==') ? 2 : clean.endsWith('=') ? 1 : 0;
    const totalLen = Math.floor((clean.length * 3) / 4) - padding;
    if (totalLen <= 0) return new ArrayBuffer(0);

    const result = new Uint8Array(totalLen);
    const CHUNK_SIZE = 65536; // 64KB multiple of 4
    let destOffset = 0;

    for (let i = 0; i < clean.length; i += CHUNK_SIZE) {
      const chunkStr = clean.substring(i, Math.min(i + CHUNK_SIZE, clean.length));
      const binChunk = atob(chunkStr);
      for (let j = 0; j < binChunk.length && destOffset < totalLen; j++) {
        result[destOffset++] = binChunk.charCodeAt(j);
      }
    }
    return result.buffer;
  } catch (err) {
    console.error('Error decoding base64:', err);
    return new ArrayBuffer(0);
  }
}

async function base64ToArrayBufferAsync(base64: string): Promise<ArrayBuffer> {
  return base64ToArrayBuffer(base64);
}

/**
 * Exports all encrypted items and cryptographic salt to a Blob in streaming chunks
 */
export async function exportEncryptedBackupBlob(
  onProgress?: (current: number, total: number) => void
): Promise<Blob> {
  const db = await openVaultDatabase();
  const blobParts: BlobPart[] = [];

  const saltHex = localStorage.getItem('vault_salt') || '';
  const testIvHex = localStorage.getItem('vault_test_iv') || '';
  const testCipherHex = localStorage.getItem('vault_test_cipher') || '';
  const customFolders = {
    photo: getSafeFolders('photo'),
    video: getSafeFolders('video'),
    audio: getSafeFolders('audio'),
    app: getSafeFolders('app'),
  };

  blobParts.push('{\n');
  blobParts.push('  "version": 3,\n');
  blobParts.push('  "appName": "Bóveda Segura - Fotos, Videos y Apps",\n');
  blobParts.push(`  "exportedAt": ${JSON.stringify(new Date().toISOString())},\n`);
  blobParts.push(`  "saltHex": ${JSON.stringify(saltHex)},\n`);
  blobParts.push(`  "testIvHex": ${JSON.stringify(testIvHex)},\n`);
  blobParts.push(`  "testCipherHex": ${JSON.stringify(testCipherHex)},\n`);
  blobParts.push(`  "customFolders": ${JSON.stringify(customFolders)},\n`);
  blobParts.push('  "items": [\n');

  // Pre-flight key list query (uses very little memory)
  const keys = await new Promise<IDBValidKey[]>((resolve) => {
    const transaction = db.transaction([STORE_NAME], 'readonly');
    const store = transaction.objectStore(STORE_NAME);
    const req = store.getAllKeys();
    req.onsuccess = () => resolve(req.result || []);
    req.onerror = () => resolve([]);
  });

  let firstItem = true;
  const totalCount = keys.length;

  // Process and stringify each item sequentially, opening fresh small transactions
  for (let i = 0; i < totalCount; i++) {
    const id = keys[i];

    if (onProgress) {
      onProgress(i + 1, totalCount);
    }

    const raw = await new Promise<any>((resolve) => {
      const transaction = db.transaction([STORE_NAME], 'readonly');
      const store = transaction.objectStore(STORE_NAME);
      const req = store.get(id);
      req.onsuccess = () => resolve(req.result);
      req.onerror = () => resolve(null);
    });

    if (!raw) continue;

    try {
      const name = raw.name || `item_${raw.id}`;
      const mimeType = raw.mimeType || (raw.category === 'video' ? 'video/mp4' : 'image/jpeg');
      const category = raw.category || detectCategory(name, mimeType);

      // Memory-efficient native base64 conversion
      const base64Data = raw.data ? await arrayBufferToBase64Async(raw.data) : '';
      
      // IMMEDIATELY dereference raw.data to free binary memory from V8 heap
      raw.data = null;

      const itemObj = {
        id: raw.id,
        timestamp: raw.timestamp || Date.now(),
        name,
        mimeType,
        size: raw.size || 0,
        iv: Array.isArray(raw.iv) ? raw.iv : (raw.iv ? Array.from(raw.iv) : []),
        base64Data,
        album: raw.album || 'General',
        isFavorite: !!raw.isFavorite,
        category,
      };

      if (!firstItem) {
        blobParts.push(',\n');
      }
      blobParts.push('    ' + JSON.stringify(itemObj));
      firstItem = false;
    } catch (err) {
      console.error('Error exporting item details:', err);
    }

    // Force full yield to event loop and Garbage Collection between files
    await new Promise((r) => setTimeout(r, 30));
  }

  blobParts.push('\n  ]\n}');

  return new Blob(blobParts, { type: 'application/json' });
}

/**
 * Attempts to repair truncated JSON files if missing closing brackets
 */
export function parseWithAutoRepair(jsonStr: string): any {
  let trimmed = jsonStr.trim();
  if (!trimmed) {
    throw new Error('El archivo de copia de seguridad está completamente vacío');
  }

  // Strip Byte Order Mark (BOM) if present
  if (trimmed.charCodeAt(0) === 0xFEFF) {
    trimmed = trimmed.slice(1);
  }
  // Strip trailing null characters and spaces
  trimmed = trimmed.replace(/\0+$/, '').trim();

  // Try direct parse first in case it was just padded
  try {
    return JSON.parse(trimmed);
  } catch (err) {
    console.warn('Initial parse in auto-repair failed, attempting structural repair:', err);
  }

  // Bracket balancing scan ignoring string literals and escapes
  let inString = false;
  let escape = false;
  const stack: ('{' | '[')[] = [];

  for (let i = 0; i < trimmed.length; i++) {
    const char = trimmed[i];
    if (escape) {
      escape = false;
      continue;
    }
    if (char === '\\') {
      escape = true;
      continue;
    }
    if (char === '"') {
      inString = !inString;
      continue;
    }
    if (!inString) {
      if (char === '{') {
        stack.push('{');
      } else if (char === '[') {
        stack.push('[');
      } else if (char === '}') {
        if (stack[stack.length - 1] === '{') {
          stack.pop();
        }
      } else if (char === ']') {
        if (stack[stack.length - 1] === '[') {
          stack.pop();
        }
      }
    }
  }

  // Find the last complete item ending with '}'
  const lastItemEnd = trimmed.lastIndexOf('}');
  if (lastItemEnd > 0) {
    let repaired = trimmed.substring(0, lastItemEnd + 1).trim();
    if (repaired.endsWith(',')) {
      repaired = repaired.slice(0, -1).trim();
    }

    // Now balance the stack for this repaired prefix
    const prefixStack: ('{' | '[')[] = [];
    inString = false;
    escape = false;
    for (let i = 0; i < repaired.length; i++) {
      const char = repaired[i];
      if (escape) { escape = false; continue; }
      if (char === '\\') { escape = true; continue; }
      if (char === '"') { inString = !inString; continue; }
      if (!inString) {
        if (char === '{') prefixStack.push('{');
        else if (char === '[') prefixStack.push('[');
        else if (char === '}') { if (prefixStack[prefixStack.length - 1] === '{') prefixStack.pop(); }
        else if (char === ']') { if (prefixStack[prefixStack.length - 1] === '[') prefixStack.pop(); }
      }
    }

    // Close open structures in reverse order
    while (prefixStack.length > 0) {
      const openToken = prefixStack.pop();
      if (openToken === '{') {
        repaired += '\n}';
      } else if (openToken === '[') {
        repaired += '\n  ]';
      }
    }

    try {
      return JSON.parse(repaired);
    } catch (err2: any) {
      console.error('Failed auto-repaired JSON parse:', err2);
      throw new Error(`El archivo de copia de seguridad está dañado: ${err2?.message || 'Unexpected end of JSON input'}`);
    }
  }

  throw new Error('El archivo de copia de seguridad está dañado o incompleto (Unexpected end of JSON input)');
}

export interface BackupHeaderInfo {
  saltHex?: string;
  testIvHex?: string;
  testCipherHex?: string;
  customFolders?: any;
  hasItems: boolean;
}

/**
 * Rapidly reads only the first ~512KB of the backup file to inspect header and cryptographic tokens,
 * without loading the rest of a multi-hundred-megabyte file into memory.
 */
export async function readBackupHeaderFromBlob(blob: Blob): Promise<BackupHeaderInfo> {
  const headerSlice = blob.slice(0, Math.min(blob.size, 512 * 1024));
  const headerText = await headerSlice.text();

  const saltMatch = headerText.match(/"saltHex"\s*:\s*"([a-fA-F0-9]+)"/);
  const ivMatch = headerText.match(/"testIvHex"\s*:\s*"([a-fA-F0-9]+)"/);
  const cipherMatch = headerText.match(/"testCipherHex"\s*:\s*"([a-fA-F0-9]+)"/);

  let customFolders: any = null;
  const foldersMatch = headerText.match(/"customFolders"\s*:\s*(\{.+?\})/s);
  if (foldersMatch) {
    try {
      customFolders = JSON.parse(foldersMatch[1]);
    } catch {}
  }

  const hasItems = /"items"\s*:\s*\[|"photos"\s*:\s*\[|"data"\s*:\s*\[|"records"\s*:\s*\[|^\s*\[/m.test(headerText);

  return {
    saltHex: saltMatch?.[1],
    testIvHex: ivMatch?.[1],
    testCipherHex: cipherMatch?.[1],
    customFolders,
    hasItems,
  };
}

async function saveItemsBatch(db: IDBDatabase, records: EncryptedVaultItem[]): Promise<void> {
  if (records.length === 0) return;
  return new Promise((resolve, reject) => {
    try {
      const tx = db.transaction([STORE_NAME], 'readwrite');
      const store = tx.objectStore(STORE_NAME);
      for (const rec of records) {
        store.put(rec);
      }
      tx.oncomplete = () => resolve();
      tx.onerror = () => reject(tx.error || new Error('Error al escribir registros en IndexedDB'));
      tx.onabort = () => reject(new Error('Transacción abortada'));
    } catch (err) {
      reject(err);
    }
  });
}

/**
 * Streams and restores encrypted items directly from a Blob or File of ANY size (e.g. 548MB+)
 * using a memory-constant chunked JSON object parser. It processes items one-by-one, saves
 * each to IndexedDB, and releases memory immediately, avoiding V8 string length limits and mobile crashes.
 */
export async function importEncryptedBackupBlobStream(
  blob: Blob,
  onProgress?: (loadedBytes: number, totalBytes: number, itemCount: number) => void,
  overwriteCryptographicTokens: boolean = false
): Promise<{ importedCount: number }> {
  // First, verify header tokens if present
  const header = await readBackupHeaderFromBlob(blob);

  if (header.saltHex && header.testIvHex && header.testCipherHex) {
    if (overwriteCryptographicTokens || !localStorage.getItem('vault_salt')) {
      localStorage.setItem('vault_salt', header.saltHex);
      localStorage.setItem('vault_test_iv', header.testIvHex);
      localStorage.setItem('vault_test_cipher', header.testCipherHex);
    }
  }

  if (header.customFolders) {
    try {
      if (Array.isArray(header.customFolders.photo)) localStorage.setItem('vault_subfolders_photo', JSON.stringify(header.customFolders.photo));
      if (Array.isArray(header.customFolders.video)) localStorage.setItem('vault_subfolders_video', JSON.stringify(header.customFolders.video));
      if (Array.isArray(header.customFolders.audio)) localStorage.setItem('vault_subfolders_audio', JSON.stringify(header.customFolders.audio));
      if (Array.isArray(header.customFolders.app)) localStorage.setItem('vault_subfolders_app', JSON.stringify(header.customFolders.app));
    } catch {}
  }

  const db = await openVaultDatabase();
  let count = 0;
  const totalBytes = blob.size;
  const CHUNK_SIZE = 1024 * 1024 * 4; // 4MB stream slices for high throughput
  let offset = 0;

  let buffer = '';
  let inArray = false;
  let depth = 0;
  let inString = false;
  let escape = false;
  let itemStart = -1;
  let cursor = 0;

  let pendingBatch: EncryptedVaultItem[] = [];
  const decoder = new TextDecoder('utf-8', { fatal: false });

  while (offset < totalBytes || cursor < buffer.length) {
    // Read next slice into buffer if needed
    if (offset < totalBytes && buffer.length - cursor < CHUNK_SIZE) {
      const nextOffset = Math.min(offset + CHUNK_SIZE, totalBytes);
      const slice = blob.slice(offset, nextOffset);
      const chunkText = decoder.decode(await slice.arrayBuffer(), { stream: nextOffset < totalBytes });
      offset = nextOffset;
      buffer += chunkText;
    }

    if (!inArray) {
      const arrayKeyMatches = [
        buffer.indexOf('"items":'),
        buffer.indexOf('"photos":'),
        buffer.indexOf('"data":'),
        buffer.indexOf('"records":'),
      ].filter((idx) => idx !== -1);

      if (arrayKeyMatches.length > 0) {
        const earliestKeyIdx = Math.min(...arrayKeyMatches);
        const arrIdx = buffer.indexOf('[', earliestKeyIdx);
        if (arrIdx !== -1) {
          buffer = buffer.substring(arrIdx + 1);
          cursor = 0;
          inArray = true;
        } else {
          if (offset >= totalBytes) break;
          cursor = buffer.length;
        }
      } else {
        const arrIdx = buffer.indexOf('[');
        if (arrIdx !== -1 && !buffer.slice(0, arrIdx).includes('{')) {
          buffer = buffer.substring(arrIdx + 1);
          cursor = 0;
          inArray = true;
        } else {
          if (offset >= totalBytes) break;
          cursor = buffer.length;
        }
      }
    }

    if (inArray) {
      while (cursor < buffer.length) {
        const c = buffer[cursor];
        if (escape) {
          escape = false;
          cursor++;
          continue;
        }
        if (c === '\\') {
          escape = true;
          cursor++;
          continue;
        }
        if (c === '"') {
          inString = !inString;
          cursor++;
          continue;
        }
        if (!inString) {
          if (c === '{') {
            if (depth === 0) itemStart = cursor;
            depth++;
          } else if (c === '}') {
            depth--;
            if (depth === 0 && itemStart !== -1) {
              const itemJson = buffer.substring(itemStart, cursor + 1);
              try {
                const item = JSON.parse(itemJson);
                const rawBase64 = item.base64Data || item.base64 || item.content || '';

                let dataBuffer: ArrayBuffer;
                if (item.data instanceof ArrayBuffer) {
                  dataBuffer = item.data;
                } else if (rawBase64) {
                  dataBuffer = base64ToArrayBuffer(rawBase64);
                } else {
                  dataBuffer = new ArrayBuffer(0);
                }

                // Auto-generate safe fallback ID and IV
                const id = item.id || `item_${Math.random().toString(36).substring(2, 11)}_${Date.now()}`;
                let iv: number[] = [];
                if (Array.isArray(item.iv)) {
                  iv = item.iv;
                } else if (item.iv && typeof item.iv === 'object') {
                  iv = Object.values(item.iv).map(Number);
                } else if (typeof item.iv === 'string' && /^[0-9a-fA-F]+$/.test(item.iv)) {
                  iv = Array.from(hexToBuf(item.iv));
                }

                const name = item.name || 'archivo_restaurado';
                const mimeType = item.mimeType || 'application/octet-stream';
                const category: VaultItemCategory = item.category || detectCategory(name, mimeType);

                const record: EncryptedVaultItem = {
                  id,
                  timestamp: item.timestamp || Date.now(),
                  name,
                  mimeType,
                  size: item.size || dataBuffer.byteLength,
                  iv,
                  data: dataBuffer,
                  album: item.album || 'General',
                  isFavorite: !!item.isFavorite,
                  category,
                };

                pendingBatch.push(record);
                count++;

                if (pendingBatch.length >= 10) {
                  await saveItemsBatch(db, pendingBatch);
                  pendingBatch = [];
                }
              } catch (parseErr) {
                console.warn('Skipping malformed item in stream:', parseErr);
              }

              // Discard parsed object from buffer immediately to free memory
              buffer = buffer.substring(cursor + 1);
              cursor = 0;
              itemStart = -1;
              continue;
            }
          } else if (c === ']' && depth === 0) {
            // End of items array reached
            buffer = '';
            offset = totalBytes;
            break;
          }
        }
        cursor++;
      }
    }

    if (onProgress) {
      onProgress(offset, totalBytes, count);
    }

    // If offset has reached totalBytes and no item is currently being read, we can finish
    if (offset >= totalBytes && itemStart === -1 && (!inArray || cursor >= buffer.length)) {
      break;
    }

    // Cooperative yield to keep UI responsive
    await new Promise((r) => setTimeout(r, 0));
  }

  // Save any remaining items in pendingBatch
  if (pendingBatch.length > 0) {
    await saveItemsBatch(db, pendingBatch);
    pendingBatch = [];
  }

  if (onProgress) {
    onProgress(totalBytes, totalBytes, count);
  }

  return { importedCount: count };
}

/**
 * Restores items and optional cryptographic tokens from a backup file (compatibility wrapper)
 */
export async function importEncryptedBackup(
  jsonContent: string,
  onProgress?: (current: number, total: number) => void,
  overwriteCryptographicTokens: boolean = false
): Promise<{ importedCount: number }> {
  const blob = new Blob([jsonContent], { type: 'application/json' });
  return importEncryptedBackupBlobStream(
    blob,
    (loaded, total, count) => {
      if (onProgress) onProgress(count, count);
    },
    overwriteCryptographicTokens
  );
}

  useEffect(() => {
    if (isSettingsOpen && !wasSettingsOpenRef.current) {
      window.history.pushState({ appOverlay: 'settings' }, '');
      wasSettingsOpenRef.current = true;
    } else if (!isSettingsOpen && wasSettingsOpenRef.current) {
      wasSettingsOpenRef.current = false;
      if (window.history.state?.appOverlay === 'settings') {
        window.history.back();
      }
    }
  }, [isSettingsOpen]);

  useEffect(() => {
    const handlePopState = () => {
      if (isSettingsOpenRef.current) {
        wasSettingsOpenRef.current = false;
        setIsSettingsOpen(false);
      }
    };
    window.addEventListener('popstate', handlePopState);
    return () => window.removeEventListener('popstate', handlePopState);
  }, []);

  // Check initial vault existence on mount
  useEffect(() => {
    const init = async () => {
      try {
        await openVaultDatabase();
        const isInit = localStorage.getItem('vault_initialized');
        const saltHex = localStorage.getItem('vault_salt');
        const testIvHex = localStorage.getItem('vault_test_iv');
        const testCipherHex = localStorage.getItem('vault_test_cipher');

        if (isInit === 'true' && saltHex && testIvHex && testCipherHex) {
          // Siempre requerir PIN o Huella al abrir o reabrir la aplicación
          setVaultStatus('locked');
        } else {
          setVaultStatus('setup');
        }
      } catch (err) {
        console.error('Initialization error:', err);
        setVaultStatus('setup');
      }
    };

    init();
    initScreenshotBlocker();
    setScreenshotBlockToastCallback(showToast);
    initBackButtonListener();
  }, [showToast]);

  // Root back button press notice
  useEffect(() => {
    const handleRootBack = () => {
      showToast('Presiona o desliza atrás de nuevo para salir', 'info');
    };
    const handleRootExit = () => {
      showToast('Saliendo de la bóveda...', 'info');
    };
    window.addEventListener('vault_root_back_press', handleRootBack);
    window.addEventListener('vault_root_exit_allowed', handleRootExit);
    return () => {
      window.removeEventListener('vault_root_back_press', handleRootBack);
      window.removeEventListener('vault_root_exit_allowed', handleRootExit);
    };
  }, [showToast]);

  // App privacy blur shield and immediate locking when minimizing or switching apps
  useEffect(() => {
    let appStateHandle: any = null;

    const handleMinimizeOrBackground = () => {
      if ((window as any).__isPickingFile || (window as any).__isProcessingBackup) {
        return;
      }
      lockVault();
    };

    if (CapacitorApp && typeof CapacitorApp.addListener === 'function') {
      try {
        CapacitorApp.addListener('appStateChange', (state) => {
          setIsPrivacyBlurActive(!state.isActive);
          if (!state.isActive) {
            handleMinimizeOrBackground();
          }
        }).then((h) => {
          appStateHandle = h;
        }).catch(() => {});
      } catch {}
    }

    const handleDocVisibility = () => {
      const isHidden = document.visibilityState === 'hidden';
      setIsPrivacyBlurActive(isHidden);
      if (isHidden) {
        handleMinimizeOrBackground();
      }
    };

    const handleWindowBlur = () => {
      setIsPrivacyBlurActive(true);
      handleMinimizeOrBackground();
    };

    const handleWindowFocus = () => {
      setIsPrivacyBlurActive(false);
    };

    document.addEventListener('visibilitychange', handleDocVisibility);
    window.addEventListener('blur', handleWindowBlur);
    window.addEventListener('focus', handleWindowFocus);

    return () => {
      if (appStateHandle && typeof appStateHandle.remove === 'function') {
        appStateHandle.remove();
      }
      document.removeEventListener('visibilitychange', handleDocVisibility);
      window.removeEventListener('blur', handleWindowBlur);
      window.removeEventListener('focus', handleWindowFocus);
    };
  }, [lockVault]);

  // Inactivity auto-lock
  useEffect(() => {
    const mins = Number(autoLockMinutes);
    if (vaultStatus !== 'unlocked' || isNaN(mins) || mins <= 0) {
      if (inactivityTimerRef.current) {
        window.clearTimeout(inactivityTimerRef.current);
        inactivityTimerRef.current = null;
      }
      return;
    }

    const resetTimer = () => {
      if (inactivityTimerRef.current) {
        window.clearTimeout(inactivityTimerRef.current);
      }
      if ((window as any).__isProcessingBackup || (window as any).__isPickingFile) {
        // Pause auto-lock completely during backup generation/restoration or file picking
        return;
      }
      inactivityTimerRef.current = window.setTimeout(() => {
        lockVault();
      }, mins * 60 * 1000);
    };

    resetTimer();

    const activityEvents = ['mousedown', 'mousemove', 'keydown', 'scroll', 'touchstart', 'pointerdown', 'click'];
    activityEvents.forEach((evt) => window.addEventListener(evt, resetTimer, { passive: true }));

    return () => {
      if (inactivityTimerRef.current) {
        window.clearTimeout(inactivityTimerRef.current);
        inactivityTimerRef.current = null;
      }
      activityEvents.forEach((evt) => window.removeEventListener(evt, resetTimer));
    };
  }, [vaultStatus, autoLockMinutes, lockVault]);

  // Robust File Picker protection: keeps auto-lock completely paused while Android File Picker is active
  useEffect(() => {
    const handleFilePickerClick = (e: MouseEvent) => {
      const target = e.target as HTMLElement;
      if (!target) return;

      const isFileInput = target.tagName === 'INPUT' && (target as HTMLInputElement).type === 'file';
      const isLabelForFile = target.closest('label') && target.closest('label')?.querySelector('input[type="file"]');

      if (isFileInput || isLabelForFile) {
        (window as any).__isPickingFile = true;
        (window as any).__filePickingStartTime = Date.now();
      }
    };

    // When the user finishes picking or cancels, reset the flag safely with a 4s grace window
    const handleFileChangeOrCancel = (e: Event) => {
      const target = e.target as HTMLElement;
      if (target && target.tagName === 'INPUT' && (target as HTMLInputElement).type === 'file') {
        setTimeout(() => {
          (window as any).__isPickingFile = false;
        }, 4000);
      }
    };

    window.addEventListener('click', handleFilePickerClick, true);
    window.addEventListener('change', handleFileChangeOrCancel, true);
    window.addEventListener('cancel', handleFileChangeOrCancel, true);

    return () => {
      window.removeEventListener('click', handleFilePickerClick, true);
      window.removeEventListener('change', handleFileChangeOrCancel, true);
      window.removeEventListener('cancel', handleFileChangeOrCancel, true);
    };
  }, []);

  // Lock on tab visibility change, minimize or pagehide
  useEffect(() => {
    if (!lockOnTabSwitch) return;

    const handleVisibilityChange = () => {
      // Ignore if user is actively picking files or processing backup
      if ((window as any).__isPickingFile || (window as any).__isProcessingBackup) {
        return;
      }

      if (document.visibilityState === 'hidden' && vaultStatus === 'unlocked') {
        // Bloquear inmediatamente al minimizar o pasar a segundo plano
        lockVault();
      }
    };

    const handlePageHide = () => {
      if ((window as any).__isPickingFile || (window as any).__isProcessingBackup) {
        return;
      }
      if (vaultStatus === 'unlocked') {
        lockVault();
      }
    };

    document.addEventListener('visibilitychange', handleVisibilityChange);
    window.addEventListener('pagehide', handlePageHide);
    return () => {
      document.removeEventListener('visibilitychange', handleVisibilityChange);
      window.removeEventListener('pagehide', handlePageHide);
    };
  }, [lockOnTabSwitch, vaultStatus, lockVault]);

  // Panic key: Escape locks immediately
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape' && vaultStatus === 'unlocked' && !isSettingsOpen) {
        lockVault();
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [vaultStatus, isSettingsOpen, lockVault]);

  // Reset Vault Handler
  const handleResetVault = async () => {
    try {
      await clearVaultDatabase();
      try {
        sessionStorage.removeItem('vault_session_pin');
      } catch {}
      localStorage.removeItem('vault_salt');
      localStorage.removeItem('vault_test_iv');
      localStorage.removeItem('vault_test_cipher');
      localStorage.removeItem('vault_initialized');
      localStorage.removeItem('vault_pin_length');

      setMasterKey(null);
      setIsSettingsOpen(false);
      setVaultStatus('setup');
      showToast('Bóveda restablecida por completo', 'alert');
    } catch (err) {
      console.error('Error resetting vault:', err);
      showToast('Error al restablecer la bóveda', 'error');
    }
  };

  // Preference change listeners
  const handleAutoLockChange = (mins: number) => {
    setAutoLockMinutes(mins);
    localStorage.setItem('vault_autolock_mins', mins.toString());
  };

  const handleLockOnTabSwitchChange = (enabled: boolean) => {
    setLockOnTabSwitch(enabled);
    localStorage.setItem('vault_lock_on_tab_switch', enabled ? 'true' : 'false');
  };

  return (
    <div className="min-h-screen bg-[#090d16] text-slate-100 flex flex-col font-sans select-none w-full max-w-full overflow-x-hidden">
      {/* Toast Notification Layer */}
      <ToastContainer toasts={toasts} />

      {/* Screen Routing */}
      {vaultStatus === 'checking' && (
        <div className="min-h-screen flex items-center justify-center">
          <div className="w-8 h-8 rounded-full border-2 border-indigo-500 border-t-transparent animate-spin" />
        </div>
      )}

      {vaultStatus === 'setup' && (
        <SetupVaultView
          onSetupComplete={(key, pin) => handleUnlock(key, pin)}
          showToast={showToast}
        />
      )}

      {vaultStatus === 'locked' && (
        <LockVaultView
          onUnlock={handleUnlock}
          onResetVaultRequest={handleResetVault}
          showToast={showToast}
        />
      )}

      {vaultStatus === 'unlocked' && masterKey && (
        <>
          <MainVaultView
            masterKey={masterKey}
            sessionPin={sessionPin}
            onLock={lockVault}
            onOpenSettings={() => setIsSettingsOpen(true)}
            showToast={showToast}
            refreshTrigger={vaultRefreshTrigger}
            autoLockMinutes={autoLockMinutes}
            onAutoLockChange={handleAutoLockChange}
            lockOnTabSwitch={lockOnTabSwitch}
            onLockOnTabSwitchChange={handleLockOnTabSwitchChange}
            onMasterKeyUpdated={(newKey) => {
              setMasterKey(newKey);
              setVaultRefreshTrigger((prev) => prev + 1);
            }}
            onResetVault={handleResetVault}
          />

          <SettingsModal
            isOpen={isSettingsOpen}
            onClose={() => setIsSettingsOpen(false)}
            masterKey={masterKey}
            sessionPin={sessionPin}
            onMasterKeyUpdated={(newKey) => {
              setMasterKey(newKey);
              setVaultRefreshTrigger((prev) => prev + 1);
            }}
            autoLockMinutes={autoLockMinutes}
            onAutoLockChange={handleAutoLockChange}
            lockOnTabSwitch={lockOnTabSwitch}
            onLockOnTabSwitchChange={handleLockOnTabSwitchChange}
            onPhotosUpdated={() => {
              setVaultRefreshTrigger((prev) => prev + 1);
            }}
            onResetVault={handleResetVault}
            showToast={showToast}
          />
        </>
      )}

      {/* Privacy Blur Shield when minimizing app or switching tasks */}
      {isPrivacyBlurActive && (
        <div className="fixed inset-0 z-[99999] backdrop-blur-3xl bg-[#060b14]/90 flex flex-col items-center justify-center p-6 text-center select-none animate-in fade-in duration-100 pointer-events-none">
          <div className="w-20 h-20 rounded-3xl bg-[#00e5ff]/10 border border-[#00e5ff]/30 flex items-center justify-center text-[#00e5ff] mb-4 shadow-[0_0_50px_rgba(0,229,255,0.25)] animate-pulse">
            <Shield className="w-10 h-10 stroke-[2]" />
          </div>
          <h3 className="text-xl font-bold text-white tracking-tight mb-1">
            Mi Bóveda Protegida
          </h3>
          <p className="text-xs text-slate-400 font-medium">
            Contenido privado oculto por seguridad
          </p>
        </div>
      )}
    </div>
  );
}
